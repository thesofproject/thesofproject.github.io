185static inline void *sof_ctx_alloc(struct mod_alloc_ctx *ctx, uint32_t flags,
186 size_t size, size_t alignment)
187{
188 if (!ctx || !ctx->vreg)
189 return sof_heap_alloc(ctx ? ctx->heap : NULL, flags, size, alignment);
190
191 if (flags & SOF_MEM_FLAG_COHERENT)
192 return vregion_alloc_coherent_align(ctx->vreg, size, alignment);
193
194 return vregion_alloc_align(ctx->vreg, size, alignment);
195}
196
205static inline void *sof_ctx_zalloc(struct mod_alloc_ctx *ctx, uint32_t flags,
206 size_t size, size_t alignment)
207{
208 void *ptr = sof_ctx_alloc(ctx, flags, size, alignment);
209
210 if (ptr)
211 memset(ptr, 0, size);
212
213 return ptr;
214}
215
221static inline void sof_ctx_free(struct mod_alloc_ctx *ctx, void *ptr)
222{
223 if (!ptr)
224 return;
225
226 if (ctx && ctx->vreg)
227 vregion_free(ctx->vreg, ptr);
228 else
229 sof_heap_free(ctx ? ctx->heap : NULL, ptr);
230}
