212static inline void *sof_ctx_alloc(struct mod_alloc_ctx *ctx, uint32_t flags,
213 size_t size, size_t alignment)
214{
215 if (!ctx || !ctx->vreg)
216 return sof_heap_alloc(ctx ? ctx->heap : NULL, flags, size, alignment);
217
218 if (flags & SOF_MEM_FLAG_COHERENT)
219 return vregion_alloc_coherent_align(ctx->vreg, size, alignment);
220
221 return vregion_alloc_align(ctx->vreg, size, alignment);
222}

232static inline void *sof_ctx_zalloc(struct mod_alloc_ctx *ctx, uint32_t flags,
233 size_t size, size_t alignment)
234{
235 void *ptr = sof_ctx_alloc(ctx, flags, size, alignment);
236
237 if (ptr)
238 memset(ptr, 0, size);
239
240 return ptr;
241}

248static inline void sof_ctx_free(struct mod_alloc_ctx *ctx, void *ptr)
249{
250 if (!ptr)
251 return;
252
253 if (ctx && ctx->vreg)
254 vregion_free(ctx->vreg, ptr);
255 else
256 sof_heap_free(ctx ? ctx->heap : NULL, ptr);
257}
