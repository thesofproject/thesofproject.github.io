270static inline void *sof_ctx_alloc(struct mod_alloc_ctx *ctx, uint32_t flags,
271 size_t size, size_t alignment)
272{
273 if (!ctx || !ctx->vreg)
274 return sof_heap_alloc(ctx ? ctx->heap : NULL, flags, size, alignment);
275
276 if (flags & SOF_MEM_FLAG_COHERENT)
277 return vregion_alloc_coherent_align(ctx->vreg, size, alignment);
278
279 return vregion_alloc_align(ctx->vreg, size, alignment);
280}

293static inline void *sof_ctx_zalloc(struct mod_alloc_ctx *ctx, uint32_t flags,
294 size_t size, size_t alignment)
295{
296 void *ptr = sof_ctx_alloc(ctx, flags, size, alignment);
297
298 if (ptr)
299 memset(ptr, 0, size);
300
301 return ptr;
302}

313static inline void sof_ctx_free(struct mod_alloc_ctx *ctx, void *ptr)
314{
315 if (!ptr)
316 return;
317
318 if (ctx && ctx->vreg)
319 vregion_free(ctx->vreg, ptr);
320 else
321 sof_heap_free(ctx ? ctx->heap : NULL, ptr);
322}
