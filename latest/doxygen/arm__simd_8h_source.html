21/*
22 * Branchless Saturation Primitives using ARM SSAT instruction
23 */
24static inline int16_t arm_sat_s16(int32_t val)
25{
26#if defined(__GNUC__)
27 int32_t res;
28 __asm__("ssat %0, #16, %1" : "=r"(res) : "r"(val));
29 return (int16_t)res;
30#elif defined(__SSAT)
31 return (int16_t)__SSAT(val, 16);
32#else
33 if (val > 32767)
34 return 32767;
35 if (val < -32768)
36 return -32768;
37 return (int16_t)val;
38#endif
39}

58static inline int32_t arm_sat_q31(int64_t val)
59{
60 if (__builtin_expect(val >= (int64_t)INT32_MIN && val <= (int64_t)INT32_MAX, 1))
61 return (int32_t)val;
62 return (val > INT32_MAX) ? INT32_MAX : INT32_MIN;
63}

65/*
66 * Dual 16-bit Arithmetic
67 */
68static inline uint32_t arm_qadd16(uint32_t a, uint32_t b)
69{
70#if defined(__GNUC__)
71 uint32_t res;
72 __asm__("qadd16 %0, %1, %2" : "=r"(res) : "r"(a), "r"(b));
73 return res;
74#elif defined(__QADD16)
75 return __QADD16(a, b);
76#else
77 int16_t a_l = (int16_t)(a & 0xffff);
78 int16_t a_h = (int16_t)(a >> 16);
79 int16_t b_l = (int16_t)(b & 0xffff);
80 int16_t b_h = (int16_t)(b >> 16);
81 int16_t r_l = arm_sat_s16((int32_t)a_l + b_l);
82 int16_t r_h = arm_sat_s16((int32_t)a_h + b_h);
83 return ((uint32_t)(uint16_t)r_h << 16) | (uint16_t)r_l;
84#endif
85}

159/*
160 * Signed Most Significant Word Multiply with Rounding (SMMULR)
161 * Computes: (a * b + 0x80000000) >> 32 in a single clock cycle.
162 * Ideal for Q1.31 audio scaling.
163 */
164static inline int32_t arm_smmulr(int32_t a, int32_t b)
165{
166#if defined(__GNUC__)
167 int32_t res;
168 __asm__("smmulr %0, %1, %2" : "=r"(res) : "r"(a), "r"(b));
169 return res;
170#else
171 return (int32_t)(((int64_t)a * b + (1LL << 31)) >> 32);
172#endif
173}

175/*
176 * 64-bit Multiply-Accumulate (SMLAL)
177 * Computes: acc += (int64_t)a * b in 1 cycle throughput.
178 */
179static inline int64_t arm_smlal(int64_t acc, int32_t a, int32_t b)
180{
181#if defined(__GNUC__)
182 uint32_t lo = (uint32_t)acc;
183 int32_t hi = (int32_t)(acc >> 32);
184 __asm__("smlal %0, %1, %2, %3" : "+r"(lo), "+r"(hi) : "r"(a), "r"(b));
185 return ((int64_t)hi << 32) | lo;
186#else
187 return acc + ((int64_t)a * b);
188#endif
189}

216/*
217 * Pack Halfwords (PKHBT / PKHTB)
218 */
219#if defined(__GNUC__)
220#define arm_pkhbt(a, b, lsl_shift) \
221__extension__ ({ \
222 uint32_t _res; \
223 uint32_t _a = (a); \
224 uint32_t _b = (b); \
225 __asm__("pkhbt %0, %1, %2, lsl %3" : "=r"(_res) : "r"(_a), "r"(_b), "I"(lsl_shift)); \
226 _res; \
227})
228#elif defined(__PKHBT)
229#define arm_pkhbt(a, b, lsl_shift) __PKHBT(a, b, lsl_shift)
230#else
231static inline uint32_t arm_pkhbt(uint32_t a, uint32_t b, uint32_t lsl_shift)
232{
233 return (a & 0xffff) | ((b << lsl_shift) & 0xffff0000);
234}
235#endif
