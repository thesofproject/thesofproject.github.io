119/*
120 * attach a secondary buffer (any type) before buffer (when at_input == true) or behind a buffer
121 *
122 * before buffer (at_input == true):
123 * 2.0 mod ==> (sink_API) secondary buffer ==>
124 * ==> comp_buffer (audio_stream or source API) ==> 1.0 mod
125 *
126 * after buffer (at_input == false):
127 * 1.0 mod ==> (audio_stream or sink API) ==> comp_buffer ==>
128 * ==> secondary buffer(source API) == 2.0 mod
129 *
130 * If a secondary buffer is attached, it replaces source or sink interface of audio_stream
131 * allowing the module connected to it using all properties of secondary buffer (like
132 * lockless cross-core connection in case of ring_buffer etc.) keeping legacy interface
133 * to other modules
134 *
135 * buffer_sync_secondary_buffer must be called every 1 ms to move data to/from
136 * secondary buffer to comp_buffer
137 *
138 * @param buffer pointer to a buffer
139 * @param at_input true indicates that a secondary buffer is located at data input, replacing
140 * sink API of audio_stream
141 * false indicates that a secondary buffer is located at data output, replacing
142 * source API of audio_stream
143 * @param secondary_buffer pointer to a buffer to be attached
144 *
145 * to be removed when hybrid buffers are no longer needed
146 */
147int audio_buffer_attach_secondary_buffer(struct sof_audio_buffer *buffer, bool at_input,
148 struct sof_audio_buffer *secondary_buffer);
149
150/*
151 * move data from/to secondary buffer, must be called periodically as described above
152 *
153 * @param buffer pointer to a buffer
154 * @param limit data copy limit. Indicates maximum amount of data that will be moved from/to
155 * secondary buffer in an operation
156 *
157 * to be removed when hybrid buffers are no longer needed
158 */
159int audio_buffer_sync_secondary_buffer(struct sof_audio_buffer *buffer, size_t limit);
