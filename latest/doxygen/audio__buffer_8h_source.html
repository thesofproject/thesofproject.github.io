125/*
126 * attach a secondary buffer (any type) before buffer (when at_input == true) or behind a buffer
127 *
128 * before buffer (at_input == true):
129 * 2.0 mod ==> (sink_API) secondary buffer ==>
130 * ==> comp_buffer (audio_stream or source API) ==> 1.0 mod
131 *
132 * after buffer (at_input == false):
133 * 1.0 mod ==> (audio_stream or sink API) ==> comp_buffer ==>
134 * ==> secondary buffer(source API) == 2.0 mod
135 *
136 * If a secondary buffer is attached, it replaces source or sink interface of audio_stream
137 * allowing the module connected to it using all properties of secondary buffer (like
138 * lockless cross-core connection in case of ring_buffer etc.) keeping legacy interface
139 * to other modules
140 *
141 * buffer_sync_secondary_buffer must be called every 1 ms to move data to/from
142 * secondary buffer to comp_buffer
143 *
144 * @param buffer pointer to a buffer
145 * @param at_input true indicates that a secondary buffer is located at data input, replacing
146 * sink API of audio_stream
147 * false indicates that a secondary buffer is located at data output, replacing
148 * source API of audio_stream
149 * @param secondary_buffer pointer to a buffer to be attached
150 *
151 * to be removed when hybrid buffers are no longer needed
152 */
153int audio_buffer_attach_secondary_buffer(struct sof_audio_buffer *buffer, bool at_input,
154 struct sof_audio_buffer *secondary_buffer);
155
156/*
157 * move data from/to secondary buffer, must be called periodically as described above
158 *
159 * @param buffer pointer to a buffer
160 * @param limit data copy limit. Indicates maximum amount of data that will be moved from/to
161 * secondary buffer in an operation
162 *
163 * to be removed when hybrid buffers are no longer needed
164 */
165int audio_buffer_sync_secondary_buffer(struct sof_audio_buffer *buffer, size_t limit);
