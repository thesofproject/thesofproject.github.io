435static inline uint32_t
436audio_stream_get_avail_bytes(const struct audio_stream *stream)
437{
438 /*
439 * In case of underrun-permitted stream, report buffer full instead of
440 * empty. This way, any data present in such stream is processed at
441 * regular pace, but buffer will never be seen as completely empty by
442 * clients, and in turn will not cause underrun/XRUN.
443 */
444 if (stream->runtime_stream_params.underrun_permitted)
445 return stream->avail != 0 ? stream->avail : stream->size;
446
447 return stream->avail;
448}

653static inline void audio_stream_produce(struct audio_stream *buffer, uint32_t bytes)
654{
655 buffer->w_ptr = audio_stream_wrap(buffer,
656 (char *)buffer->w_ptr + bytes);
657
658 /* "overwrite" old data in circular wrap case */
659 if (bytes > audio_stream_get_free_bytes(buffer))
660 buffer->r_ptr = buffer->w_ptr;
661
662 /* calculate available bytes */
663 if (buffer->r_ptr < buffer->w_ptr)
664 buffer->avail = (char *)buffer->w_ptr - (char *)buffer->r_ptr;
665 else if (buffer->r_ptr == buffer->w_ptr)
666 buffer->avail = buffer->size; /* full */
667 else
668 buffer->avail = buffer->size -
669 ((char *)buffer->r_ptr - (char *)buffer->w_ptr);
670
671 /* calculate free bytes */
672 buffer->free = buffer->size - buffer->avail;
673}

680static inline void audio_stream_consume(struct audio_stream *buffer, uint32_t bytes)
681{
682 buffer->r_ptr = audio_stream_wrap(buffer,
683 (char *)buffer->r_ptr + bytes);
684
685 /* calculate available bytes */
686 if (buffer->r_ptr < buffer->w_ptr)
687 buffer->avail = (char *)buffer->w_ptr - (char *)buffer->r_ptr;
688 else if (buffer->r_ptr == buffer->w_ptr)
689 buffer->avail = 0; /* empty */
690 else
691 buffer->avail = buffer->size -
692 ((char *)buffer->r_ptr - (char *)buffer->w_ptr);
693
694 /* calculate free bytes */
695 buffer->free = buffer->size - buffer->avail;
696}

704static inline void audio_stream_sync_to_hw(struct audio_stream *stream,
705 const struct dma_status *dma_status)
706{
707 uintptr_t buf_start = (uintptr_t)audio_stream_get_addr(stream);
708
709 /* Note: It is assumed here that dma_status values are reported as bytes. However,
710 * this is actually platform-specific. Although unlikely, they could be, for example,
711 * words on some particular platform, and changes should be made here to address
712 * such case.
713 */
714
715 assert(dma_status->write_position < audio_stream_get_size(stream));
716 assert(dma_status->read_position < audio_stream_get_size(stream));
717 assert(dma_status->pending_length <= audio_stream_get_size(stream));
718 assert(dma_status->free <= audio_stream_get_size(stream));
719
720 audio_stream_set_wptr(stream, (void *)(buf_start + dma_status->write_position));
721 audio_stream_set_rptr(stream, (void *)(buf_start + dma_status->read_position));
722 audio_stream_set_avail(stream, dma_status->pending_length);
723 audio_stream_set_free(stream, dma_status->free);
724}

982static inline int audio_stream_set_zero(struct audio_stream *buffer, uint32_t bytes)
983{
984 uint32_t head_size = bytes;
985 uint32_t tail_size = 0;
986
987 /* check for overrun */
988 if (audio_stream_get_free_bytes(buffer) < bytes)
989 return 1;
990
991 /* check for potential wrap */
992 if ((char *)buffer->w_ptr + bytes > (char *)buffer->end_addr) {
993 head_size = (char *)buffer->end_addr - (char *)buffer->w_ptr;
994 tail_size = bytes - head_size;
995 }
996
997 memset(buffer->w_ptr, 0, head_size);
998 if (tail_size)
999 memset(buffer->addr, 0, tail_size);
1000
1001 return 0;
1002}

1027static inline int audio_stream_fmt_conversion(enum ipc4_bit_depth depth,
1028 enum ipc4_bit_depth valid,
1029 enum sof_ipc_frame *frame_fmt,
1030 enum sof_ipc_frame *valid_fmt,
1031 enum ipc4_sample_type type)
1032{
1033 int ret = -EINVAL;
1034 *frame_fmt = SOF_IPC_FRAME_INVALID;
1035 *valid_fmt = SOF_IPC_FRAME_INVALID;
1036
1037 switch (type) {
1038 case IPC4_TYPE_FLOAT:
1039#ifdef CONFIG_FORMAT_FLOAT
1040 if (depth == 32 && valid == 32) {
1041 *frame_fmt = SOF_IPC_FRAME_FLOAT;
1042 *valid_fmt = SOF_IPC_FRAME_FLOAT;
1043 ret = 0;
1044 }
1045#endif
1046 break;
1047 case IPC4_TYPE_MSB_INTEGER:
1048 case IPC4_TYPE_LSB_INTEGER:
1049 case IPC4_TYPE_SIGNED_INTEGER:
1050 if (depth == 24 && valid == 24) {
1051#ifdef CONFIG_FORMAT_S24_3LE
1052 *frame_fmt = SOF_IPC_FRAME_S24_3LE;
1053 *valid_fmt = SOF_IPC_FRAME_S24_3LE;
1054 ret = 0;
1055#endif
1056 } else {
1057 /* IPC4_DEPTH_16BIT (16) <---> SOF_IPC_FRAME_S16_LE (0)
1058 * IPC4_DEPTH_24BIT (24) <---> SOF_IPC_FRAME_S24_4LE (1)
1059 * IPC4_DEPTH_32BIT (32) <---> SOF_IPC_FRAME_S32_LE (2)
1060 */
