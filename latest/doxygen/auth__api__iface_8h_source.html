67struct auth_api {
68 /* Interface to initialize authentication API and context.
69 * Parameters:
70 * ctx - pointer to the context instance of type auth_api_ctx.
71 * scratch_buff - pointer to scratch buffer.
72 * Scratch buffer must be located in L2 Local Memory (SHA Engine limitation).
73 * Caller is responsible to power up necessary L2 Local Memory banks.
74 * Address alignment must correspond to SHA384_IO_BUF_ALIGNMENT.
75 * scratch_buff_size – size must be the same as AUTH_SCRATCH_BUFF_SZ.
76 * Return value:
77 * ADSP_SUCCESS - successful initialization.
78 */
79 int (*init)(struct auth_api_ctx *ctx, void *scratch_buff, size_t scratch_buff_size,
80 enum auth_image_type image_type);
81
82 /* Interface to cleanup authentication API.
83 * Parameters:
84 * ctx - pointer to the context instance of type AuthApiCtx.
85 */
86 void (*cleanup)(struct auth_api_ctx *ctx);
87
88 /* Interface for initiating signed FW image (async) authentication process.
89 * Parameters:
90 * ctx - pointer to the context instance of type AuthApiCtx.
91 * chunk - pointer to the chunk of signed FW image.
92 * chunk_size - chunk size in bytes.
93 * phase - authentication phase.
94 * Must corresponds to one of the AuthPhase values.
95 * In case of one time FW authentication, where signed FW image size must be
96 * less or equal to scratch_buff_size, the caller must pass AUTH_PHASE_LAST.
97 * Return value: ADSP_SUCCESS when authentication process has been initiated
98 * successfully, or one of ADSP_FLV_* error codes in case of failure.
99 */
100 int (*init_auth_proc)(struct auth_api_ctx *ctx, const void *chunk, size_t chunk_size,
101 enum auth_phase phase);
102
103 /* Interface to return if authentication process is busy.
104 * Parameters:
105 * ctx - pointer to the context instance of type AuthApiCtx.
106 * This function can be used for authentication process synchronization.
107 * Return value: true if authentication process is busy.
108 */
109 bool (*busy)(struct auth_api_ctx *ctx);
110
111 /* Interface to return authentication result
112 * Parameters:
113 * ctx - pointer to the context instance of type AuthApiCtx.
114 * Return value:
115 * AUTH_NOT_COMPLETED - authentication is not completed,
116 * AUTH_IMAGE_TRUSTED - authentication completed and signed FW image is
117 * trusted,
118 * AUTH_IMAGE_UNTRUSTED - authentication completed, but signed FW image is
119 * untrusted.
120 */
121 enum auth_result (*result)(struct auth_api_ctx *ctx);
122
123 /* Interface to register status/error code logger.
124 * Parameters:
125 * ctx - pointer to the context instance of type AuthApiCtx.
126 * sts_logger - pointer to status logger.
127 * Return value: ADSP_SUCCESS when logger has been registered successfully.
128 */
129 int (*register_status_logger)(struct auth_api_ctx *ctx,
130 struct status_logger_ctx *status_logger);
131
132 /* Interface to unregister status/error code logger.
133 * Parameters:
134 * ctx - pointer to the context instance of type AuthApiCtx.
135 */
136 void (*unregister_status_logger)(struct auth_api_ctx *ctx);
137};
