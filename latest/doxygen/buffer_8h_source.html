127 * this is a legacy component connector for pipeline 1.0

287/*
288 * Attach a new buffer at the beginning of the list. Note, that "head" must
289 * really be the head of the list, not a list head within another buffer. We
290 * don't synchronise its cache, so it must not be embedded in an object, using
291 * the coherent API. The caller takes care to protect list heads.
292 *
293 * Returns -EINVAL if the buffer is already linked in this direction
294 * (re-attaching would create a self-loop and corrupt the list).
295 */
296int buffer_attach(struct comp_buffer *buffer, struct list_item *head, int dir);
297
298/*
299 * Detach a buffer from anywhere in the list. "head" is again the head of the
300 * list, we need it to verify, whether this buffer was the first or the last on
301 * the list. Again it is assumed that the head's cache doesn't need to be
302 * synchronized. The caller takes care to protect list heads.
303 */
304void buffer_detach(struct comp_buffer *buffer, struct list_item *head, int dir);

309 comp_buffer_get_sink_component(buffer) :
310 comp_buffer_get_source_component(buffer);
311 return comp;
