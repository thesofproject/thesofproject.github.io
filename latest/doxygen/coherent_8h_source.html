33/*
34 * The coherent API allows optimized access to memory by multiple cores, using
35 * cache, taking care about coherence. The intended use is to let cores acquire
36 * ownership of such shared objects, use them, and then release them, possibly
37 * to be re-acquired by other cores. Such shared objects must only be accessed
38 * via this API. It's designed to be primarily used with dynamically allocated
39 * objects because of their well-defined life span. It can also be used with
40 * objects from .data or .bss sections but greater care must be takenwith them
41 * to strictly follow the API flow.
42 *
43 * The API assumes, that in the beginning no core has cache lines associated
44 * with the memory area, used with it. That is true for dynamically allocated
45 * memory, because when such memory is freed, its cache is invalidated - as long
46 * as that memory was never accessed by other cores, except by using this API.
47 * The first call must be coherent_init(), which initializes the header. If the
48 * object will be used by multiple cores, next coherent_shared() must be called.
49 * After that to use that memory, coherent_acquire() must be called, which
50 * acquires ownership of the object and returns a cached address of the memory.
51 * After that the user can perform cached access to the memory. To release the
52 * memory, coherent_release() must be called. The only time when the memory is
53 * accessed using cache is between those two calls, so only when releasing the
54 * memory we have to write back and invalidate caches to make sure, that next
55 * time we acquire this memory, our uncached header access will not be
56 * overwritten! When memory is not needed any more, typically before freeing the
57 * memory, coherent_free() should be called.
58 *
59 * This structure needs to be embedded at the start of any container to ensure
60 * container object cache alignment and to minimise non cache access when
61 * acquiring ownership.
62 *
63 * This structure must not be accessed outside of these APIs.
64 * The shared flag is only set at coherent init and thereafter it's RO.
65 */

151__must_check static inline struct coherent __sparse_cache *coherent_acquire(struct coherent *c,
152 const size_t size)
153{
154 struct coherent __sparse_cache *cc = uncache_to_cache(c);
155
156 /* assert if someone passes a cache/local address in here. */
157 ADDR_IS_COHERENT(c);
158 CHECK_ATOMIC(c);
159
160 /* access the shared coherent object */
161 if (c->shared) {
162 CHECK_COHERENT_CORE(c);
163
164 c->key = k_spin_lock(&c->lock);
165
166 /*
167 * FIXME: This is wrong. dcache_invalidate_region() only makes
168 * sense if we assume, that dirty cache lines might exist for
169 * this object. But in that case those lines could be written
170 * back here thus overwriting either user data, or the coherent
171 * header or both. When coherent_acquire() is called it must be
172 * guaranteed that the object isn't in cache. Before it is
173 * acquired no cached access to it is allowed. This has to be
174 * fixed here and on multiple further occasions below.
175 */
176
177 /* invalidate local copy */
178 dcache_invalidate_region(cc, size);
179 }
180
181 /* client can now use cached object safely */
182 return cc;
183}

194static inline void coherent_release(struct coherent __sparse_cache *c,
195 const size_t size)
196{
197 /* assert if someone passes a coherent address in here. */
198 ADDR_IS_INCOHERENT(c);
199 CHECK_ATOMIC(c);
200
201 /* access the local copy of object */
202 if (c->shared) {
203 struct coherent *uc = cache_to_uncache(c);
204 CHECK_COHERENT_CORE(c);
205
206 /* wtb and inv local data to coherent object */
207 dcache_writeback_invalidate_region(c, size);
208
209 /* unlock on uncache alias */
210 k_spin_unlock(&uc->lock, uc->key);
211 }

224static inline void *__coherent_init(size_t offset, const size_t size)
225{
226 /*
227 * Allocate an object with an uncached alias but align size on a cache-
228 * line boundary to avoid sharing a cache line with the adjacent
229 * allocation
230 */
231 void *object = rzalloc(SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
232 ALIGN_UP(size, PLATFORM_DCACHE_ALIGN));
233 struct coherent *c;
234
235 if (!object)
236 return NULL;
237
238 c = (struct coherent *)((uint8_t *)object + offset);
239
240 /* TODO static assert if we are not cache aligned */
241 k_spinlock_init(&c->lock);
242 c->sleep_allowed = false;
243 c->shared = false;
244 c->core = cpu_get_id();
245 list_init(&c->list);
246 /* inv local data to coherent object */
247 dcache_invalidate_region(uncache_to_cache(object), size);
248
249 return object;
250}
