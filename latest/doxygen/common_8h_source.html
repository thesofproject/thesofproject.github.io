145/* Compile-time assertion.
146 *
147 * The first, typedef-based solution silently succeeds with variables,
148 * for instance STATIC_ASSERT(n == 42, always_succeeds) when 'n' is a
149 * variable in a function. The second, array-based solution is not
150 * fooled by variables but it increases the .bss size at the -O0
151 * optimization level (no difference with any real -O). As we're often
152 * short on space, use the typedef-based version by default. If you're
153 * afraid that some assertions are being fooled by variables then
154 * temporarily and locally switch to the second one.
155 */
156#define STATIC_ASSERT(COND, MESSAGE) \
157 __attribute__((unused)) \
158 typedef char _CONCAT(assertion_failed_, MESSAGE)[(COND) ? 1 : -1]

160/* Allows checking preprocessor symbols in compile-time.
161 * Returns true for config with value 1, false for undefined or any other value.
162 *
163 * Expression like (CONFIG_MY ? a : b) rises compilation error when CONFIG_MY
164 * is undefined, (IS_ENABLED(CONFIG_MY) ? a : b) can be used instead of it.
165 *
166 * It is intended to be used with Kconfig's bool configs - it should rise
167 * error for other types, except positive integers, but it shouldn't be
168 * used with them.
169 */
170#ifndef __ZEPHYR__ /* Already present and compatible via Zephyr headers */
171#define IS_ENABLED(config) IS_ENABLED_STEP_1(config)
172#define IS_ENABLED_DUMMY_1 0,
173#define IS_ENABLED_STEP_1(config) IS_ENABLED_STEP_2(IS_ENABLED_DUMMY_ ## config)
174#define IS_ENABLED_STEP_2(values) IS_ENABLED_STEP_3(values 1, 0)
175#define IS_ENABLED_STEP_3(ignore, value, ...) (!!(value))
