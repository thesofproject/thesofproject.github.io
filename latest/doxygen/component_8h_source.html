289 * bind call will be called twice for every component, first when binding a data source,
290 * than when binding data sink.
291 *
292 * bind_type is indicating type of binding
293 */
294 enum bind_type bind_type;
295
296 /* pointers to sink or source API of the data provider/consumer
297 * that is being bound to the module
298 *
299 * if bind_type == COMP_BIND_TYPE_SOURCE, the pointer points to source being bound/unbound
300 * if bind_type == COMP_BIND_TYPE_SINK, the pointer points to sink being bound/unbound
301 *
302 * NOTE! As in pipeline2.0 there may be a binding between modules,
303 * without a buffer in between, it cannot be a pointer to any buffer type, module should
304 * use sink/source API
305 */
306 union {
307 struct sof_source *source;
308 struct sof_sink *sink;
309 };
310};

882static inline struct comp_dev *comp_alloc(const struct comp_driver *drv, size_t bytes)
883{
884 /*
885 * Use uncached address everywhere to access components to rule out
886 * multi-core failures. TODO: verify if cached alias may be used in some cases
887 */
888 struct comp_dev *dev = sof_heap_alloc(drv->user_heap,
889 SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
890 bytes, 0);
891
892 if (!dev)
893 return NULL;
894
895 memset(dev, 0, sizeof(*dev));
896 comp_init(drv, dev, bytes);
897
898 return dev;
899}

1047/* \brief Set component period frames */
1048static inline void component_set_nearest_period_frames(struct comp_dev *current,
1049 uint32_t rate)
1050{
1051 uint64_t frames;
1052
1053 /* Sample rate is in Hz and period in microseconds.
1054 * As we don't have floats use scale divider 1000000.
1055 * Also integer round up the result.
1056 * dma buffer size should align with 32bytes which can't be
1057 * compatible with current 45K adjustment. 48K is a suitable
1058 * adjustment.
1059 */
1060
1061 switch (rate) {
1062 case 44100:
1063 rate = 48000;
1064 break;
1065 case 88200:
1066 rate = 96000;
1067 break;
1068 case 176400:
1069 rate = 192000;
1070 break;
1071 default:
1072 break;
1073 }
1074
1075 frames = SOF_DIV_ROUND_UP((uint64_t)rate * current->period, 1000000);
1076
