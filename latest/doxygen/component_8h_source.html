823static inline struct comp_dev *comp_alloc(const struct comp_driver *drv, size_t bytes)
824{
825 /*
826 * Use uncached address everywhere to access components to rule out
827 * multi-core failures. TODO: verify if cached alias may be used in some cases
828 */
829 struct comp_dev *dev = sof_heap_alloc(drv->user_heap,
830 SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
831 bytes, 0);
832
833 if (!dev)
834 return NULL;
835
836 memset(dev, 0, sizeof(*dev));
837 comp_init(drv, dev, bytes);
838
839 return dev;
840}

1005static inline void component_set_nearest_period_frames(struct comp_dev *current,
1006 uint32_t rate)
1007{
1008 uint64_t frames;
1009
1010 /* Sample rate is in Hz and period in microseconds.
1011 * As we don't have floats use scale divider 1000000.
1012 * Also integer round up the result.
1013 * dma buffer size should align with 32bytes which can't be
1014 * compatible with current 45K adjustment. 48K is a suitable
1015 * adjustment.
1016 */
1017
1018 switch (rate) {
1019 case 44100:
1020 rate = 48000;
1021 break;
1022 case 88200:
1023 rate = 96000;
1024 break;
1025 case 176400:
1026 rate = 192000;
1027 break;
1028 default:
1029 break;
1030 }
1031
1032 frames = SOF_DIV_ROUND_UP((uint64_t)rate * current->period, 1000000);
1033
1034 assert(frames <= UINT_MAX);
1035 current->frames = (uint32_t)frames;
1036}

1048static inline void comp_underrun(struct comp_dev *dev,
1049 struct comp_buffer *source,
1050 uint32_t copy_bytes)
1051{
1052 LOG_MODULE_DECLARE(component, CONFIG_SOF_LOG_LEVEL);
1053
1054 int32_t bytes = (int32_t)audio_stream_get_avail_bytes(&source->stream) -
1055 copy_bytes;
1056
1057 comp_err(dev, "comp_underrun(): dev->comp.id = %u, source->avail = %u, copy_bytes = %u",
1058 dev_comp_id(dev),
1059 audio_stream_get_avail_bytes(&source->stream),
1060 copy_bytes);
1061
1062 pipeline_xrun(dev->pipeline, dev, bytes);
1063}
