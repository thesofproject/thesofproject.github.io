284struct bind_info {

288 /* type of binding
289 * bind call will be called twice for every component, first when binding a data source,
290 * than when binding data sink.
291 *
292 * bind_type is indicating type of binding
293 */
294 enum bind_type bind_type;
295
296 /* pointers to sink or source API of the data provider/consumer
297 * that is being bound to the module
298 *
299 * if bind_type == COMP_BIND_TYPE_SOURCE, the pointer points to source being bound/unbound
300 * if bind_type == COMP_BIND_TYPE_SINK, the pointer points to sink being bound/unbound
301 *
302 * NOTE! As in pipeline2.0 there may be a binding between modules,
303 * without a buffer in between, it cannot be a pointer to any buffer type, module should
304 * use sink/source API
305 */
306 union {
307 struct sof_source *source;
308 struct sof_sink *sink;
309 };
310};

892static inline struct comp_dev *comp_alloc(const struct comp_driver *drv, size_t bytes)
893{
894 /*
895 * Use uncached address everywhere to access components to rule out
896 * multi-core failures. TODO: verify if cached alias may be used in some cases
897 */
898 struct comp_dev *dev = sof_heap_alloc(drv->user_heap,
899 SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
900 bytes, 0);
901
902 if (!dev)
903 return NULL;
904
905 memset(dev, 0, sizeof(*dev));
906 comp_init(drv, dev, bytes);
907
908 return dev;
909}

1057/* \brief Set component period frames */
1058static inline void component_set_nearest_period_frames(struct comp_dev *current,
1059 uint32_t rate)
1060{
1061 uint64_t frames;
1062
1063 /* Sample rate is in Hz and period in microseconds.
1064 * As we don't have floats use scale divider 1000000.
1065 * Also integer round up the result.
1066 * dma buffer size should align with 32bytes which can't be
1067 * compatible with current 45K adjustment. 48K is a suitable
1068 * adjustment.
1069 */
1070
1071 switch (rate) {
1072 case 44100:
1073 rate = 48000;
1074 break;
1075 case 88200:
1076 rate = 96000;
1077 break;
