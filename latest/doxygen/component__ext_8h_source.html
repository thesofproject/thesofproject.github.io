94static inline int comp_params(struct comp_dev *dev,
95 struct sof_ipc_stream_params *params)
96{
97 int ret = 0;
98
99 if (dev->is_shared && !cpu_is_me(dev->ipc_config.core)) {
100 ret = comp_params_remote(dev, params);
101 } else {
102 if (dev->drv->ops.params) {
103 ret = dev->drv->ops.params(dev, params);
104 } else {
105 /* not defined, run the default handler */
106 ret = comp_verify_params(dev, 0, params);
107
108/* BugLink: https://github.com/zephyrproject-rtos/zephyr/issues/43786
109 * TODO: Remove this once the bug gets fixed.
110 */
111#ifndef __ZEPHYR__
112 if (ret < 0)
113 comp_err(dev, "pcm params verification failed");
114#endif
115 }
116 }
117
118 return ret;
119}

147static inline int comp_cmd(struct comp_dev *dev, int cmd, void *data,
148 int max_data_size)
149{
150 LOG_MODULE_DECLARE(component, CONFIG_SOF_LOG_LEVEL);
151
152 struct sof_ipc_ctrl_data *cdata = ASSUME_ALIGNED(data, 4);
153
154 if (cmd == COMP_CMD_SET_DATA &&
155 (cdata->data->magic != SOF_ABI_MAGIC ||
156 SOF_ABI_VERSION_INCOMPATIBLE(SOF_ABI_VERSION, cdata->data->abi))) {
157 comp_err(dev, "comp_cmd(): invalid version, data->magic = %u, data->abi = %u",
158 cdata->data->magic, cdata->data->abi);
159 return -EINVAL;
160 }
161
162 if (dev->drv->ops.cmd)
163 return dev->drv->ops.cmd(dev, cmd, data, max_data_size);
164
165 return -EINVAL;
166}

190static inline int comp_trigger_local(struct comp_dev *dev, int cmd)
191{
192 int ret;
193
194 ret = dev->drv->ops.trigger(dev, cmd);
195 if (ret)
196 return ret;
197
198 /* start a thread in case of shared component or DP scheduling */
199 if (dev->task) {
200 /* schedule or cancel task */
201 switch (cmd) {
202 case COMP_TRIGGER_START:
203 ret = schedule_task(dev->task, 0, dev->period);
204 break;
205 case COMP_TRIGGER_RELEASE:
206 case COMP_TRIGGER_XRUN:
207 case COMP_TRIGGER_PAUSE:
208 case COMP_TRIGGER_STOP:
209 break;
210 }
211 }
212
213 return ret;
214}

285static inline int comp_ipc4_get_attribute_remote(struct comp_dev *dev, uint32_t type,
286 void *value)
287{
288 struct ipc4_base_module_cfg *base_cfg;
289 struct get_attribute_remote_payload payload = {};
290 struct idc_msg msg = { IDC_MSG_GET_ATTRIBUTE,
291 IDC_EXTENSION(dev->ipc_config.id), dev->ipc_config.core,
292 sizeof(payload), &payload};
293 int ret;
294
295 /* Only COMP_ATTR_BASE_CONFIG is supported for remote access */
296 if (type != COMP_ATTR_BASE_CONFIG)
297 return -EINVAL;
298
299 base_cfg = rzalloc(SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT, sizeof(*base_cfg));
300 if (!base_cfg)
301 return -ENOMEM;
302
303 payload.type = type;
304 payload.value = base_cfg;
305
306 ret = idc_send_msg(&msg, IDC_BLOCKING);
307
308 if (ret == 0)
309 memcpy_s(value, sizeof(struct ipc4_base_module_cfg),
310 base_cfg, sizeof(struct ipc4_base_module_cfg));
311
312 rfree(base_cfg);
313 return ret;
314}
315#endif /* CONFIG_IPC_MAJOR_4 */

472static inline int comp_get_requested_state(int cmd)
473{
474 int state = COMP_STATE_INIT;
475
476 switch (cmd) {
477 case COMP_TRIGGER_START:
478 case COMP_TRIGGER_RELEASE:
479 state = COMP_STATE_ACTIVE;
480 break;
481 case COMP_TRIGGER_PREPARE:
482 case COMP_TRIGGER_STOP:
483 state = COMP_STATE_PREPARE;
484 break;
485 case COMP_TRIGGER_PAUSE:
486 state = COMP_STATE_PAUSED;
487 break;
488 case COMP_TRIGGER_XRUN:
489 case COMP_TRIGGER_RESET:
490 state = COMP_STATE_READY;
491 break;
492 case COMP_TRIGGER_PRE_START:
493 case COMP_TRIGGER_PRE_RELEASE:
494 state = COMP_STATE_PRE_ACTIVE;
495 break;
496 }
497
498 return state;
499}

657static inline bool comp_same_dir(struct comp_dev *comp, enum sof_ipc_stream_direction dir)
658{
659 int end_type = comp_get_endpoint_type(comp->pipeline->sink_comp);
660
661 if (dir == SOF_IPC_STREAM_PLAYBACK && end_type != COMP_ENDPOINT_DAI)
662 return false;
663 if (dir == SOF_IPC_STREAM_CAPTURE && end_type != COMP_ENDPOINT_HOST)
664 return false;
665 return true;
666}
