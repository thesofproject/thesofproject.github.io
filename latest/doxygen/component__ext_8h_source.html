73static inline int comp_params(struct comp_dev *dev,
74 struct sof_ipc_stream_params *params)
75{
76 int ret = 0;
77
78 if (dev->is_shared && !cpu_is_me(dev->ipc_config.core)) {
79 ret = comp_params_remote(dev, params);
80 } else {
81 if (dev->drv->ops.params) {
82 ret = dev->drv->ops.params(dev, params);
83 } else {
84 /* not defined, run the default handler */
85 ret = comp_verify_params(dev, 0, params);
86
87/* BugLink: https://github.com/zephyrproject-rtos/zephyr/issues/43786
88 * TODO: Remove this once the bug gets fixed.
89 */
90#ifndef __ZEPHYR__
91 if (ret < 0)
92 comp_err(dev, "pcm params verification failed");
93#endif
94 }
95 }
96
97 return ret;
98}

113static inline int comp_cmd(struct comp_dev *dev, int cmd, void *data,
114 int max_data_size)
115{
116 LOG_MODULE_DECLARE(component, CONFIG_SOF_LOG_LEVEL);
117
118 struct sof_ipc_ctrl_data *cdata = ASSUME_ALIGNED(data, 4);
119
120 if (cmd == COMP_CMD_SET_DATA &&
121 (cdata->data->magic != SOF_ABI_MAGIC ||
122 SOF_ABI_VERSION_INCOMPATIBLE(SOF_ABI_VERSION, cdata->data->abi))) {
123 comp_err(dev, "comp_cmd(): invalid version, data->magic = %u, data->abi = %u",
124 cdata->data->magic, cdata->data->abi);
125 return -EINVAL;
126 }
127
128 if (dev->drv->ops.cmd)
129 return dev->drv->ops.cmd(dev, cmd, data, max_data_size);
130
131 return -EINVAL;
132}

147static inline int comp_trigger_local(struct comp_dev *dev, int cmd)
148{
149 int ret;
150
151 ret = dev->drv->ops.trigger(dev, cmd);
152 if (ret)
153 return ret;
154
155 /* start a thread in case of shared component or DP scheduling */
156 if (dev->task) {
157 /* schedule or cancel task */
158 switch (cmd) {
159 case COMP_TRIGGER_START:
160 ret = schedule_task(dev->task, 0, dev->period);
161 break;
162 case COMP_TRIGGER_RELEASE:
163 case COMP_TRIGGER_XRUN:
164 case COMP_TRIGGER_PAUSE:
165 case COMP_TRIGGER_STOP:
166 break;
167 }
168 }
169
170 return ret;
171}

180 ret = comp_trigger_remote(dev, cmd);
181 else
182 ret = comp_trigger_local(dev, cmd);
183
184 return ret;

214static inline int comp_ipc4_get_attribute_remote(struct comp_dev *dev, uint32_t type,
215 void *value)
216{
217 struct ipc4_base_module_cfg *base_cfg;
218 struct get_attribute_remote_payload payload = {};
219 struct idc_msg msg = { IDC_MSG_GET_ATTRIBUTE,
220 IDC_EXTENSION(dev->ipc_config.id), dev->ipc_config.core,
221 sizeof(payload), &payload};
222 int ret;
223
224 /* Only COMP_ATTR_BASE_CONFIG is supported for remote access */
225 if (type != COMP_ATTR_BASE_CONFIG)
226 return -EINVAL;
227
228 base_cfg = rzalloc(SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT, sizeof(*base_cfg));
229 if (!base_cfg)
230 return -ENOMEM;
231
232 payload.type = type;
233 payload.value = base_cfg;
234
235 ret = idc_send_msg(&msg, IDC_BLOCKING);
236
237 if (ret == 0)
238 memcpy_s(value, sizeof(struct ipc4_base_module_cfg),
239 base_cfg, sizeof(struct ipc4_base_module_cfg));
240
241 rfree(base_cfg);
242 return ret;
243}
244#endif /* CONFIG_IPC_MAJOR_4 */

362static inline int comp_get_requested_state(int cmd)
363{
364 int state = COMP_STATE_INIT;
365
366 switch (cmd) {
367 case COMP_TRIGGER_START:
368 case COMP_TRIGGER_RELEASE:
369 state = COMP_STATE_ACTIVE;
370 break;
371 case COMP_TRIGGER_PREPARE:
372 case COMP_TRIGGER_STOP:
373 state = COMP_STATE_PREPARE;
374 break;
375 case COMP_TRIGGER_PAUSE:
376 state = COMP_STATE_PAUSED;
377 break;
378 case COMP_TRIGGER_XRUN:
379 case COMP_TRIGGER_RESET:
380 state = COMP_STATE_READY;
381 break;
382 case COMP_TRIGGER_PRE_START:
383 case COMP_TRIGGER_PRE_RELEASE:
384 state = COMP_STATE_PRE_ACTIVE;
385 break;
386 }
387
388 return state;
389}

511static inline bool comp_same_dir(struct comp_dev *comp, enum sof_ipc_stream_direction dir)
512{
513 int end_type = comp_get_endpoint_type(comp->pipeline->sink_comp);
514
515 if (dir == SOF_IPC_STREAM_PLAYBACK && end_type != COMP_ENDPOINT_DAI)
516 return false;
517 if (dir == SOF_IPC_STREAM_CAPTURE && end_type != COMP_ENDPOINT_HOST)
518 return false;
519 return true;
520}
