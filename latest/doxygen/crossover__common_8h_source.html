51static inline void crossover_reset_state_lr4(struct processing_module *mod,
52 struct iir_state_df1 *lr4)
53{
54 mod_free(mod, lr4->coef);
55 mod_free(mod, lr4->delay);
56
57 lr4->coef = NULL;
58 lr4->delay = NULL;
59}

65static inline void crossover_reset_state_ch(struct processing_module *mod,
66 struct crossover_state *ch_state)
67{
68 int i;
69
70 for (i = 0; i < CROSSOVER_MAX_LR4; i++) {
71 crossover_reset_state_lr4(mod, &ch_state->lowpass[i]);
72 crossover_reset_state_lr4(mod, &ch_state->highpass[i]);
73 }
74}

79static inline crossover_split crossover_find_split_func(int32_t num_sinks)
80{
81 if (num_sinks < CROSSOVER_2WAY_NUM_SINKS ||
82 num_sinks > CROSSOVER_4WAY_NUM_SINKS)
83 return NULL;
84 // The functions in the map are offset by 2 indices.
85 return crossover_split_fnmap[num_sinks - CROSSOVER_2WAY_NUM_SINKS];
86}
