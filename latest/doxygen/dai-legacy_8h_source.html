513static inline int dai_get_info(struct dai *dai, int info)
514{
515 int ret;
516
517 switch (info) {
518 case DAI_INFO_TYPE:
519 ret = dai->drv->type;
520 break;
521 case DAI_INFO_DMA_CAPS:
522 ret = dai->drv->dma_caps;
523 break;
524 case DAI_INFO_DMA_DEV:
525 ret = dai->drv->dma_dev;
526 break;
527 default:
528 ret = -EINVAL;
529 break;
530 }
531
532 return ret;
533}
