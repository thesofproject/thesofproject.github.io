76/* DMA API
77 * Programming flow is :-
78 *
79 * 1) dma_channel_get()
80 * 2) notifier_register()
81 * 3) dma_set_config()
82 * 4) dma_start()
83 * ... DMA now running ...
84 * 5) dma_stop()
85 * 6) dma_stop_delayed()
86 * 7) dma_channel_put()
87 */

89static inline struct dma_chan_data *dma_channel_get_legacy(struct dma *dma,
90 int req_channel)
91{
92 if (!dma || !dma->ops || !dma->ops->channel_get)
93 return NULL;
94
95 struct dma_chan_data *chan = dma->ops->channel_get(dma, req_channel);
96
97 return chan;
98}
