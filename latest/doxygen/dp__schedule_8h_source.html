108/*
109 * Keeps the scheduler_dp_thread_ipc() flow simple - just one call that does all
110 * the IPC-message specific parameter packing internally. This is slightly
111 * suboptimal because IPC parameters first have to be collected in this
112 * structure and then packed in DP-accessible memory inside
113 * scheduler_dp_thread_ipc(). This could be split into two levels, by adding
114 * IPC-specific functions like ipc_flatten_pipeline_state() and similar, but
115 * that would add multiple functions to the API.
116 */
117union scheduler_dp_thread_ipc_param {
118 const struct bind_info *bind_data;
119 /* SOF_IPC4_MOD_INIT_INSTANCE: ext_data points to the caller's stack frame and is
120 * only valid until scheduler_dp_thread_ipc() returns; ipc_thread_flatten() copies
121 * it by value into DP-thread-accessible memory before the DP thread runs init().
122 */
123 const struct module_ext_init_data *ext_data;
124 struct {
125 unsigned int trigger_cmd;
126 enum ipc4_pipeline_state state;
127 int n_sources;
128 struct sof_source **sources;
129 int n_sinks;
130 struct sof_sink **sinks;
131 } pipeline_state;
132};
