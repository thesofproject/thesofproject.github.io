16/*
17 * ESP32-P4 Processor Instruction Extensions (PIE / Xespv2p2) 128-bit SIMD.
18 *
19 * To ensure 100% build compatibility with the upstream Zephyr SDK GCC
20 * (which lacks custom -march=xesppie assembler recognition), all PIE
21 * instructions are implemented using native 32-bit machine word directives (.word).
22 *
23 * Register mapping conventions:
24 * q0..q7 : 128-bit vector registers (16x8b, 8x16b, or 4x32b)
25 * QACC : 256-bit accumulator pair (QACC_H / QACC_L)
26 * SAR : 6-bit Shift Amount Register
27 * a0 : Source pointer (incremented by 16 bytes on vld.128.ip)
28 * a2 : Broadcast scalar / control value
29 */
30
31/* Zeroing */
32#define ESP_PIE_ZERO_Q0() __asm__ volatile (".word 0x0040005b") /* esp.zero.q q0 */
33#define ESP_PIE_ZERO_Q1() __asm__ volatile (".word 0x004000db") /* esp.zero.q q1 */
34#define ESP_PIE_ZERO_Q2() __asm__ volatile (".word 0x0040015b") /* esp.zero.q q2 */
35#define ESP_PIE_ZERO_Q3() __asm__ volatile (".word 0x004001db") /* esp.zero.q q3 */
36#define ESP_PIE_ZERO_QACC() __asm__ volatile (".word 0x0000025b") /* esp.zero.qacc */

87/* Shift Amount Register (SAR) and CFG configuration */
88static inline void esp_pie_set_sar(uint32_t sar)
89{
90 register uint32_t sar_reg asm("a0") = sar;
91 __asm__ volatile (".word 0x90b1005f" : "+r"(sar_reg) : : "memory");
92}
93#define ESP_PIE_SET_SAR(sar) esp_pie_set_sar(sar)
94#define ESP_PIE_SET_SAR_A0() esp_pie_set_sar(15)
95#define ESP_PIE_SET_SAR_A2() __asm__ volatile (".word 0x90b2005f" :: "r"(sar_a2)) /* esp.movx.w.sar a2 */
96#define ESP_PIE_SET_CFG_A0() __asm__ volatile (".word 0x90d1005f" :: "r"(cfg_a0)) /* esp.movx.w.cfg a0 */
