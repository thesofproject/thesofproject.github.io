26static inline int32_t sat_int32(int64_t x)
27{
28 /* Shift left by 32 bits with saturation */
29 ae_f64 shifted = AE_SLAI64S(x, 32);
30
31 /* Shift right by 32 bits */
32 return (int32_t)AE_MOVINT32_FROMINT64(AE_SRAI64(shifted, 32));
33}

103static inline int16_t sat_int16(int32_t x)
104{
105 /* Move 32-bit value to ae_int32x2 type */
106 ae_int32x2 d0 = AE_MOVDA32(x);
107
108 /* Saturate both values to 16-bit and pack them */
109 ae_int16x4 result = AE_SAT16X4(d0, d0);
110
111 /* Use AE_MOVAD16_0 to extract the first 16-bit value */
112 return AE_MOVAD16_0(result);
113}

127static inline ae_int16x4 vec_sat_int16x4(int32_t x, int32_t y, int32_t z, int32_t q)
128{
129 /* Move four 32-bit values to ae_int32x2 type */
130 ae_int32x2 d0 = AE_MOVDA32X2(x, y);
131 ae_int32x2 d1 = AE_MOVDA32X2(z, q);

146static inline int8_t sat_int8(int32_t x)
147{
148 /* Shift left by 24 bits */
149 ae_f32x2 a_i = AE_SLAI32S(x, 24);
150
151 /* Shift right by 24 bits to sign-extend the 8-bit value */
152 a_i = AE_SRAI32(a_i, 24);
153
154 /* Extract the lower 8 bits as an int8_t */
155 return (int8_t)AE_MOVAD32_L(a_i);
156}

168static inline ae_int32x2 vec_sat_int8x2(int32_t x, int32_t y)
169{
170 /* Move two 32-bit values to ae_f32x2 type */
171 ae_f32x2 d0 = AE_MOVDA32X2(x, y);
172
173 /* Shift left by 24 bits and then right by 24 bits to sign-extend and saturate */
174 return AE_F32X2_SRAI(AE_F32X2_SLAIS(d0, 24), 24);
175}
