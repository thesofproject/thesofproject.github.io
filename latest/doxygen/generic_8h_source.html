48/*
49 * \brief Macro to declare a module adapter component.
50 * \param adapter - name of the module.
51 * \param uuid - UUID of the module.
52 * \param tr - trace context for the module.
53 *
54 * This macro declares a module component with the specified name, UUID, and trace context.
55 * It initializes the component module structure with the appropriate type, UID, and
56 * struct module_interface operations.
57 */
58#define DECLARE_MODULE_ADAPTER(adapter, uuid, tr) \
59static const struct comp_driver comp_##adapter##_module = { \
60 .type = SOF_COMP_MODULE_ADAPTER, \
61 .uid = SOF_RT_UUID(uuid), \
62 SET_MODULE_ADAPTER_TCTX(tr) \
63 .ops = { \
64 .create = module_adapter_new, \
65 .prepare = module_adapter_prepare, \
66 .params = module_adapter_params, \
67 .copy = module_adapter_copy, \
68 IPC_MOD_CMD(module_adapter_cmd) \
69 .trigger = module_adapter_trigger, \
70 .reset = module_adapter_reset, \
71 .free = module_adapter_free, \
72 .set_large_config = module_set_large_config,\
73 .get_large_config = module_get_large_config,\
74 .get_attribute = module_adapter_get_attribute,\
75 .set_attribute = module_adapter_set_attribute,\
76 .bind = module_adapter_bind,\
77 .unbind = module_adapter_unbind,\
78 .get_total_data_processed = module_adapter_get_total_data_processed,\
79 .dai_get_hw_params = module_adapter_get_hw_params,\
80 .position = module_adapter_position,\
81 .dai_ts_config = module_adapter_ts_config_op,\
82 .dai_ts_start = module_adapter_ts_start_op,\
83 .dai_ts_stop = module_adapter_ts_stop_op,\
84 .dai_ts_get = module_adapter_ts_get_op,\
85 }, \
86 .adapter_ops = &(adapter), \
87}; \
88\
89static struct comp_driver_info comp_module_##adapter##_info = { \
90 .drv = &comp_##adapter##_module, \
91}; \
92\
93UT_STATIC void sys_comp_module_##adapter##_init(void) \
94{ \
95 comp_register(&comp_module_##adapter##_info); \
96} \
97\

325static inline struct ipc_msg *mod_ipc_msg_w_ext_init(struct processing_module *mod,
326 uint32_t header,
327 uint32_t extension,
328 uint32_t size)
329{
330 struct ipc_msg *msg;
331
332 /* msg->list is linked into ipc->msg_list and handled cross-core: must be uncached */
333 msg = mod_alloc_ext(mod, SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT, sizeof(*msg), 0);
334 if (!msg)
335 return NULL;
336
337 memset(msg, 0, sizeof(*msg));
338
339 if (size) {
340 msg->tx_data = mod_alloc_ext(mod, SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
341 size, 0);
342 if (!msg->tx_data) {
343 mod_free(mod, msg);
344 return NULL;
345 }
346 memset(msg->tx_data, 0, size);
347 }
348
349 msg->header = header;
350 msg->extension = extension;
351 msg->tx_size = size;
352 list_init(&msg->list);
353
354 return msg;
355}

358 uint32_t header, uint32_t size)

360 return mod_ipc_msg_w_ext_init(mod, header, 0, size);

420static inline
421bool generic_module_is_ready_to_process(struct processing_module *mod,
422 struct sof_source **sources,
423 int num_of_sources,
424 struct sof_sink **sinks,
425 int num_of_sinks)
426{
427 int i;
428
429 for (i = 0; i < num_of_sources; i++)
430 if (source_get_data_available(sources[i]) < source_get_min_available(sources[i]))
431 return false;
432
433 for (i = 0; i < num_of_sinks; i++)
434 if (sink_get_free_size(sinks[i]) < sink_get_min_free_space(sinks[i]))
435 return false;
436
437 return true;
438}

440static inline
441bool module_is_ready_to_process(struct processing_module *mod,
442 struct sof_source **sources,
443 int num_of_sources,
444 struct sof_sink **sinks,
445 int num_of_sinks)
446{
447 const struct module_interface *const ops = mod->dev->drv->adapter_ops;
448
449 /* LL module has to be always ready for processing */
450 if (mod->dev->ipc_config.proc_domain == COMP_PROCESSING_DOMAIN_LL)
451 return true;
452
453 if (ops->is_ready_to_process)
454 return ops->is_ready_to_process(mod, sources, num_of_sources, sinks, num_of_sinks);
455 /* default action - the module is ready if there's enough data for processing and enough
456 * space to store result. IBS/OBS as declared in init_instance
457 */
458 return generic_module_is_ready_to_process(mod, sources, num_of_sources, sinks,
459 num_of_sinks);
460}
