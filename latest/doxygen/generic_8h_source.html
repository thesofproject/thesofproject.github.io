41/*
42 * \brief Macro to declare a module adapter component.
43 * \param adapter - name of the module.
44 * \param uuid - UUID of the module.
45 * \param tr - trace context for the module.
46 *
47 * This macro declares a module component with the specified name, UUID, and trace context.
48 * It initializes the component module structure with the appropriate type, UID, and
49 * struct module_interface operations.
50 */
51#define DECLARE_MODULE_ADAPTER(adapter, uuid, tr) \
52static const struct comp_driver comp_##adapter##_module = { \
53 .type = SOF_COMP_MODULE_ADAPTER, \
54 .uid = SOF_RT_UUID(uuid), \
55 .tctx = &(tr), \
56 .ops = { \
57 .create = module_adapter_new, \
58 .prepare = module_adapter_prepare, \
59 .params = module_adapter_params, \
60 .copy = module_adapter_copy, \
61 IPC_MOD_CMD(module_adapter_cmd) \
62 .trigger = module_adapter_trigger, \
63 .reset = module_adapter_reset, \
64 .free = module_adapter_free, \
65 .set_large_config = module_set_large_config,\
66 .get_large_config = module_get_large_config,\
67 .get_attribute = module_adapter_get_attribute,\
68 .set_attribute = module_adapter_set_attribute,\
69 .bind = module_adapter_bind,\
70 .unbind = module_adapter_unbind,\
71 .get_total_data_processed = module_adapter_get_total_data_processed,\
72 .dai_get_hw_params = module_adapter_get_hw_params,\
73 .position = module_adapter_position,\
74 .dai_ts_config = module_adapter_ts_config_op,\
75 .dai_ts_start = module_adapter_ts_start_op,\
76 .dai_ts_stop = module_adapter_ts_stop_op,\
77 .dai_ts_get = module_adapter_ts_get_op,\
78 }, \
79 .adapter_ops = &(adapter), \
80}; \
81\
82static struct comp_driver_info comp_module_##adapter##_info = { \
83 .drv = &comp_##adapter##_module, \
84}; \
85\
86UT_STATIC void sys_comp_module_##adapter##_init(void) \
87{ \
88 comp_register(&comp_module_##adapter##_info); \
89} \
90\

253static inline struct ipc_msg *mod_ipc_msg_w_ext_init(struct processing_module *mod,
254 uint32_t header,
255 uint32_t extension,
256 uint32_t size)
257{
258 struct ipc_msg *msg;
259
260 msg = mod_zalloc(mod, sizeof(*msg));
261 if (!msg)
262 return NULL;
263
264 if (size) {
265 msg->tx_data = mod_zalloc(mod, size);
266 if (!msg->tx_data) {
267 mod_free(mod, msg);
268 return NULL;
269 }
270 }
271
272 msg->header = header;
273 msg->extension = extension;
274 msg->tx_size = size;
275 list_init(&msg->list);
276
277 return msg;
278}

328static inline
329bool generic_module_is_ready_to_process(struct processing_module *mod,
330 struct sof_source **sources,
331 int num_of_sources,
332 struct sof_sink **sinks,
333 int num_of_sinks)
334{
335 int i;
336
337 for (i = 0; i < num_of_sources; i++)
338 if (source_get_data_available(sources[i]) < source_get_min_available(sources[i]))
339 return false;
340
341 for (i = 0; i < num_of_sinks; i++)
342 if (sink_get_free_size(sinks[i]) < sink_get_min_free_space(sinks[i]))
343 return false;
344
345 return true;
346}

348static inline
349bool module_is_ready_to_process(struct processing_module *mod,
350 struct sof_source **sources,
351 int num_of_sources,
352 struct sof_sink **sinks,
353 int num_of_sinks)
354{
355 const struct module_interface *const ops = mod->dev->drv->adapter_ops;
356
357 /* LL module has to be always ready for processing */
358 if (mod->dev->ipc_config.proc_domain == COMP_PROCESSING_DOMAIN_LL)
359 return true;
360
361 if (ops->is_ready_to_process)
362 return ops->is_ready_to_process(mod, sources, num_of_sources, sinks, num_of_sinks);
363 /* default action - the module is ready if there's enough data for processing and enough
364 * space to store result. IBS/OBS as declared in init_instance
365 */
366 return generic_module_is_ready_to_process(mod, sources, num_of_sources, sinks,
367 num_of_sinks);
368}
