38/*
39 * These helpers are optimized for FFT calculation only.
40 * e.g. _add/sub() assume the output won't be saturate so no check needed,
41 * and _mul() assumes Q1.31 * Q1.31 so the output will be shifted to be Q1.31.
42 */
43
44static inline void icomplex32_add(const struct icomplex32 *in1, const struct icomplex32 *in2,
45 struct icomplex32 *out)
46{
47 out->real = in1->real + in2->real;
48 out->imag = in1->imag + in2->imag;
49}
50
51static inline void icomplex32_adds(const struct icomplex32 *in1, const struct icomplex32 *in2,
52 struct icomplex32 *out)
53{
54 out->real = sat_int32((int64_t)in1->real + in2->real);
55 out->imag = sat_int32((int64_t)in1->imag + in2->imag);
56}
57
58static inline void icomplex32_sub(const struct icomplex32 *in1, const struct icomplex32 *in2,
59 struct icomplex32 *out)
60{
61 out->real = in1->real - in2->real;
62 out->imag = in1->imag - in2->imag;
63}
64
65static inline void icomplex32_mul(const struct icomplex32 *in1, const struct icomplex32 *in2,
66 struct icomplex32 *out)
67{
68 out->real = ((int64_t)in1->real * in2->real - (int64_t)in1->imag * in2->imag) >> 31;
69 out->imag = ((int64_t)in1->real * in2->imag + (int64_t)in1->imag * in2->real) >> 31;
70}

72/* complex conjugate */
73static inline void icomplex32_conj(struct icomplex32 *comp)
74{
75 comp->imag = SATP_INT32((int64_t)-1 * comp->imag);
76}

78/* shift a complex n bits, n > 0: left shift, n < 0: right shift */
79static inline void icomplex32_shift(const struct icomplex32 *input, int32_t n,
80 struct icomplex32 *output)
81{
82 if (n > 0) {
83 /* need saturation handling */
84 output->real = SATP_INT32(SATM_INT32((int64_t)input->real << n));
85 output->imag = SATP_INT32(SATM_INT32((int64_t)input->imag << n));
86 } else {
87 output->real = input->real >> -n;
88 output->imag = input->imag >> -n;
89 }
90}
