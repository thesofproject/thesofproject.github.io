287static inline struct ll_schedule_domain *domain_init
288 (int type, int clk, bool synchronous,
289 const struct ll_schedule_domain_ops *ops)
290{
291 struct ll_schedule_domain *domain;
292
293#ifdef CONFIG_SOF_USERSPACE_LL
294 domain = sof_heap_alloc(zephyr_ll_user_heap(), SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT,
295 sizeof(*domain), sizeof(void *));
296 if (domain)
297 memset(domain, 0, sizeof(*domain));
298#else
299 domain = rzalloc(SOF_MEM_FLAG_KERNEL | SOF_MEM_FLAG_COHERENT, sizeof(*domain));
300#endif
301 if (!domain)
302 return NULL;
303 domain->type = type;
304 domain->clk = clk;
305 domain->synchronous = synchronous;
306 domain->full_sync = false;
307#ifdef __ZEPHYR__
308 domain->ticks_per_ms = k_ms_to_cyc_ceil64(1);
309#else
310 domain->ticks_per_ms = clock_ms_to_ticks(clk, 1);
311#endif
312 domain->ops = ops;
313 /* maximum value means no tick has been set to timer */
314 domain->next_tick = UINT64_MAX;
315 domain->new_target_tick = UINT64_MAX;
316
317#ifdef CONFIG_SOF_USERSPACE_LL
318 /* Allocate mutex dynamically for userspace access */
319 domain->lock = k_object_alloc(K_OBJ_MUTEX);
320 if (!domain->lock) {
321 sof_heap_free(zephyr_ll_user_heap(), domain);
322 return NULL;
323 }
324 k_mutex_init(domain->lock);
325#else
326 k_spinlock_init(&domain->lock);
327#endif
328 atomic_init(&domain->total_num_tasks, 0);
329 atomic_init(&domain->enabled_cores, 0);
330
331 return domain;
332}

414static inline int domain_register(struct ll_schedule_domain *domain,
415 struct task *task,
416 void (*handler)(void *arg), void *arg)
417{
418 int ret;
419
420 assert(domain->ops->domain_register);
421
422 ret = domain->ops->domain_register(domain, task, handler, arg);
423
424 if (!ret)
425 /* registered one more task, increase the count */
426 atomic_add(&domain->total_num_tasks, 1);
427
428 return ret;
429}
430
437static inline void domain_unregister(struct ll_schedule_domain *domain,
438 struct task *task, uint32_t num_tasks)
439{
440 int ret;
441
442 assert(domain->ops->domain_unregister);
443
444 /* unregistering a task, decrement the count */
445 if (task)
446 atomic_sub(&domain->total_num_tasks, 1);
447
448 /*
449 * In some cases .domain_unregister() might not return, terminating the
450 * current thread, that's why we had to update state before calling it.
451 */
452 ret = domain->ops->domain_unregister(domain, task, num_tasks);
453 if (ret < 0 && task)
454 /* Failed to unregister the domain, restore state */
455 atomic_add(&domain->total_num_tasks, 1);
456}

463static inline void domain_enable(struct ll_schedule_domain *domain, int core)
464{
465 if (!domain->enabled[core] && domain->ops->domain_enable) {
466 domain->ops->domain_enable(domain, core);
467 domain->enabled[core] = true;
468 atomic_add(&domain->enabled_cores, 1);
469 }
470}

477static inline void domain_disable(struct ll_schedule_domain *domain, int core)
478{
479 if (domain->enabled[core] && domain->ops->domain_disable) {
480 domain->ops->domain_disable(domain, core);
481 domain->enabled[core] = false;
482 atomic_sub(&domain->enabled_cores, 1);
483 }
484}
