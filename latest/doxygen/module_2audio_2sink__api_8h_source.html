358static inline int sink_bind(struct sof_sink *sink, struct processing_module *module)
359{
360 int ret = 0;
361
362 if (sink->bound_module)
363 return -EBUSY;
364
365 if (sink->ops->on_bind)
366 ret = sink->ops->on_bind(sink, module);
367
368 if (!ret)
369 sink->bound_module = module;
370
371 return ret;
372}
373
379static inline int sink_unbind(struct sof_sink *sink)
380{
381 int ret = 0;
382
383 if (!sink->bound_module)
384 return -EINVAL;
385
386 if (sink->ops->on_unbind)
387 ret = sink->ops->on_unbind(sink);
388
389 if (!ret)
390 sink->bound_module = NULL;
391
392 return ret;
393}
