310static inline int source_bind(struct sof_source *source, struct processing_module *module)
311{
312 int ret = 0;
313
314 if (source->bound_module)
315 return -EBUSY;
316
317 if (source->ops->on_bind)
318 ret = source->ops->on_bind(source, module);
319
320 if (!ret)
321 source->bound_module = module;
322
323 return ret;
324}
325
331static inline int source_unbind(struct sof_source *source)
332{
333 int ret = 0;
334
335 if (!source->bound_module)
336 return -EINVAL;
337
338 if (source->ops->on_unbind)
339 ret = source->ops->on_unbind(source);
340
341 if (!ret)
342 source->bound_module = NULL;
343
344 return ret;
345}

376static inline uint32_t source_align_frames_round_up(struct sof_source *source, uint32_t frames)
377{
378 uint16_t align = source->audio_stream_params->align_frame_cnt;
379
380 if (align <= 1)
381 return frames;
382
383 return ROUND_UP(frames, align);
384}

386static inline uint32_t source_align_frames_round_down(struct sof_source *source, uint32_t frames)
387{
388 uint16_t align = source->audio_stream_params->align_frame_cnt;
389
390 if (align <= 1)
391 return frames;
392
393 return ROUND_DOWN(frames, align);
394}

396static inline uint32_t source_align_frames_round_nearest(struct sof_source *source, uint32_t frames)
397{
398 uint16_t align = source->audio_stream_params->align_frame_cnt;
399
400 if (!align)
401 return frames;
402
403 return ROUND_DOWN(frames + (align >> 1), align);
404}
