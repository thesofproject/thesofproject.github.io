149struct ipc4_base_module_cfg_ext {
150 /* specifies number of items in input_pins array. Maximum size is 8 */
151 uint16_t nb_input_pins;
152 /* specifies number of items in output_pins array. Maximum size is 8 */
153 uint16_t nb_output_pins;
154 uint8_t reserved[12];
155 /* Specifies format of input pins followed by output pins.
156 * Pin format arrays may be non-continuous i.e. may contain pin #0 format
157 * followed by pin #2 format in case pin #1 will not be in use.
158 * FW assigned format of the pin based on pin_index, not on a position of
159 * the item in the array. Applies to both input and output pins.
160 */
161 uint8_t pin_formats[];
162} __attribute__((packed, aligned(4)));

185struct ipc4_pin_props {
