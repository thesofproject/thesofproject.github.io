257/*
258 * Using Module Config Get / Set command, host driver may send a parameter
259 * that fits into the header (a very short one), packed along with parameter id.
260 * Larger parameters require fragmentation and a series of Large Config Set
261 * commands.
262 *
263 * param_id_data specifies both ID of the parameter, defined by the module
264 * and value of the parameter.
265 * It is up to the module how to distribute bits to ID and value of the parameter.
266 * If there are more bits required than available to value, then Input Data may
267 * be used to pass the value
268 *
269 * NOTE: Module Config Get/Set commands are used internally by the driver
270 * for small parameters defined by Intel components. While all externally
271 * developed components communicates with host using Large Config commands
272 * no matter what the size of parameter is.
273 */
274struct ipc4_module_config {
275 union {
276 uint32_t dat;
277
278 struct {
279 uint32_t module_id : 16; /* module id */
280 uint32_t instance_id : 8; /* instance id */
281 /* SOF_IPC4_MOD_CONFIG_GET / SOF_IPC4_MOD_CONFIG_SET */
282 uint32_t type : 5;
283 uint32_t rsp : 1; /* SOF_IPC4_MESSAGE_DIR_MSG_REQUEST */
284 uint32_t msg_tgt : 1; /* SOF_IPC4_MESSAGE_TARGET_MODULE_MSG */
285 uint32_t _reserved_0 : 1;
286 } r;
287 } primary;
288
289 union {
290 uint32_t dat;
291
292 struct {
293 /* Param id and data */
294 uint32_t param_id_data : 30;
295 uint32_t _reserved_2 : 2;
296 } r;
297 } extension;
298} __attribute__((packed, aligned(4)));
