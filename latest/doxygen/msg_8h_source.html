49static inline struct ipc_msg *ipc_msg_w_ext_init(uint32_t header, uint32_t extension,
50 uint32_t size)
51{
52 struct ipc_msg *msg;
53
54 msg = rzalloc(SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT, sizeof(*msg));
55 if (!msg)
56 return NULL;
57
58 if (size) {
59 msg->tx_data = rzalloc(SOF_MEM_FLAG_USER | SOF_MEM_FLAG_COHERENT, size);
60 if (!msg->tx_data) {
61 rfree(msg);
62 return NULL;
63 }
64 }
65
66 msg->header = header;
67 msg->extension = extension;
68 msg->tx_size = size;
69 list_init(&msg->list);
70
71 return msg;
72}

89static inline void ipc_msg_free(struct ipc_msg *msg)
90{
91 if (!msg)
92 return;
93
94 struct ipc *ipc = ipc_get();
95 k_spinlock_key_t key;
96
97 key = k_spin_lock(&ipc->lock);
98
99 list_item_del(&msg->list);
100 rfree(msg->tx_data);
101 rfree(msg);
102
103 k_spin_unlock(&ipc->lock, key);
104}
