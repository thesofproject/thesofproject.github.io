196static inline void ipc4_notification_watchdog_init(struct ipc4_watchdog_timeout_notification *notif,
197 uint32_t core_id, bool first_timeout)
198{
199 notif->primary.dat = 0;
200 notif->extension.dat = 0;
201
202 /* ID of a core that timeouted. */
203 notif->primary.r.core_id = core_id;
204
205 /* Indicates that it was first timeout and crash dump was done */
206 notif->primary.r.first_timeout = first_timeout;
207
208 notif->primary.r.notif_type = SOF_IPC4_WATCHDOG_TIMEOUT;
209 notif->primary.r.type = SOF_IPC4_GLB_NOTIFICATION;
210 notif->primary.r.rsp = SOF_IPC4_MESSAGE_DIR_MSG_REQUEST;
211 notif->primary.r.msg_tgt = SOF_IPC4_MESSAGE_TARGET_FW_GEN_MSG;
212}
