218static inline void ipc4_notification_watchdog_init(struct ipc4_watchdog_timeout_notification *notif,
219 uint32_t core_id, bool first_timeout)
220{
221 notif->primary.dat = 0;
222 notif->extension.dat = 0;
223
224 /* ID of a core that timeouted. */
225 notif->primary.r.core_id = core_id;
226
227 /* Indicates that it was first timeout and crash dump was done */
228 notif->primary.r.first_timeout = first_timeout;
229
230 notif->primary.r.notif_type = SOF_IPC4_WATCHDOG_TIMEOUT;
231 notif->primary.r.type = SOF_IPC4_GLB_NOTIFICATION;
232 notif->primary.r.rsp = SOF_IPC4_MESSAGE_DIR_MSG_REQUEST;
233 notif->primary.r.msg_tgt = SOF_IPC4_MESSAGE_TARGET_FW_GEN_MSG;
234}
