51/* This is a divide function that returns ceil of the quotient.
52 * E.g. ceil_divide(9, 3) returns 3, ceil_divide(10, 3) returns 4.
53 */
54static inline int ceil_divide(int a, int b)
55{
56 int c;
57
58 c = a / b;
59
60 /* First, we check whether the signs of the params are different.
61 * If they are, we already know the result is going to be negative and
62 * therefore, is going to be already rounded up (truncated).
63 *
64 * If the signs are the same, we check if there was any remainder in
65 * the division by multiplying the number back.
66 */
67 if (!((a ^ b) & (1U << ((sizeof(int) * 8) - 1))) && c * b != a)
68 c++;
69
70 return c;
71}
