57/* This is a divide function that returns ceil of the quotient.
58 * E.g. ceil_divide(9, 3) returns 3, ceil_divide(10, 3) returns 4.
59 */
60static inline int ceil_divide(int a, int b)
61{
62 int c;
63
64 c = a / b;
65
66 /* First, we check whether the signs of the params are different.
67 * If they are, we already know the result is going to be negative and
68 * therefore, is going to be already rounded up (truncated).
69 *
70 * If the signs are the same, we check if there was any remainder in
71 * the division by multiplying the number back.
72 */
73 if (!((a ^ b) & (1U << ((sizeof(int) * 8) - 1))) && c * b != a)
74 c++;
75
76 return c;
77}
