93static inline pcm_converter_func
94pcm_get_conversion_function(enum sof_ipc_frame in,
95 enum sof_ipc_frame out)
96{
97 uint32_t i;
98
99 for (i = 0; i < pcm_func_count; i++) {
100 if (in != pcm_func_map[i].source)
101 continue;
102 if (out != pcm_func_map[i].sink)
103 continue;
104
105 return pcm_func_map[i].func;
106 }
107
108 return NULL;
109}

160static inline pcm_converter_func
161pcm_get_conversion_vc_function(enum sof_ipc_frame in_bits,
162 enum sof_ipc_frame valid_in_bits,
163 enum sof_ipc_frame out_bits,

167{
168 uint32_t i;

172 continue;
173 if (valid_in_bits != pcm_func_vc_map[i].valid_src_bits)
174 continue;
175 if (out_bits != pcm_func_vc_map[i].sink)
176 continue;
177 if (valid_out_bits != pcm_func_vc_map[i].valid_sink_bits)
178 continue;
179
180 return pcm_func_vc_map[i].func;
181 }
182
183 return NULL;
184}
