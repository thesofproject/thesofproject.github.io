95static inline pcm_converter_func
96pcm_get_conversion_function(enum sof_ipc_frame in,
97 enum sof_ipc_frame out)
98{
99 uint32_t i;
100
101 for (i = 0; i < pcm_func_count; i++) {
102 if (in != pcm_func_map[i].source)
103 continue;
104 if (out != pcm_func_map[i].sink)
105 continue;
106
107 return pcm_func_map[i].func;
108 }
109
110 return NULL;
111}
