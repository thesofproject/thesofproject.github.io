145struct io_perf_data_item {
146 /* ID of interface */
147 uint32_t id : 8;
148 /* Instance of interface / bus */
149 uint32_t instance : 8;
150 /* I/O direction from ACE perspective: 0 - Input, 1 - Output */
151 uint32_t direction : 1;
152 /* I/O state: 0 - powered down / disabled, 1 - powered up / enabled */
153 uint32_t state : 1;
154 /* Power Mode: 0 - D0, 1 - D0ix (clock gating enabled), */
155 uint32_t power_mode : 2;
156 uint32_t rsvd : 11;
157 /* The component still exists (0) or has been already deleted (1) */
158 uint32_t is_removed : 1;
159 /* Performance data */
160 /*
161 * I/O (id) - ID - Units - Description
162 *
163 * Host IPC - 0 - Count - Counter of Host IPC messages incoming and outcoming
164 * IDC - 1 - Count - Counter of IDC messages incoming and outcoming per DSP core
165 * DMIC - 2 - Bytes - Counter of bytes transferred over DMIC interface
166 * I2S - 3 - Bytes - Counter of bytes transferred over I2S interface
167 * SoundWire - 4 - Bytes - Counter of bytes transferred over SoundWire interface
168 * HD/A - 5 - Bytes - Counter of bytes transferred over HD/A interface
169 * USB - 6 - Bytes - Counter of bytes transferred over USB interface
170 * GPIO - 7 - Count - Counter of GPIO interrupts or triggers
171 * I2C - 8 - Bytes - Counter of bytes transferred over I2C interface
172 * I3C - 9 - Bytes - Counter of bytes transferred over I3C interface
173 * I3C interrupt - 10 - Bytes - Counter of I3C interrupts
174 * UART - 11 - Bytes - Counter of bytes transferred over UART interface
175 * SPI - 12 - Bytes - Counter of bytes transferred over SPI interface
176 * CSI-2 - 13 - Bytes - Counter of bytes transferred over CSI-2 interface
177 * DTF - 14 - Bytes - Counter of bytes transferred over DTF interface
178 */
179 uint64_t data;
180} __packed;
