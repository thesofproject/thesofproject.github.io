244static inline void *scheduler_list_get_data(struct schedulers *schedulers, uint16_t type)
245{
246 struct schedule_data *sch;
247 struct list_item *slist;
248
249 if (!schedulers)
250 return NULL;
251
252 list_for_item(slist, &schedulers->list) {
253 sch = container_of(slist, struct schedule_data, list);
254 if (type == sch->type)
255 return sch->data;
256 }
257
258 return NULL;
259}

332static inline int schedule_task_before(struct task *task, uint64_t start,
333 uint64_t period, struct task *before)
