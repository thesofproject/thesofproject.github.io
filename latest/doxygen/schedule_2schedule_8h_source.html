189static inline void *scheduler_list_get_data(struct schedulers *schedulers, uint16_t type)
190{
191 struct schedule_data *sch;
192 struct list_item *slist;
193
194 if (!schedulers)
195 return NULL;
196
197 list_for_item(slist, &schedulers->list) {
198 sch = container_of(slist, struct schedule_data, list);
199 if (type == sch->type)
200 return sch->data;
201 }
202
203 return NULL;
204}
