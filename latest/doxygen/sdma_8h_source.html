11#define SDMA_MC0PTR 0x0000 /* platform 0 control block */
12#define SDMA_INTR 0x0004 /* Active interrupts, W1C */
13#define SDMA_STOP_STAT 0x0008 /* Channel stop/status, W1C */
14#define SDMA_HSTART 0x000C /* Channel start */
15
16/* Prevent hardware requests from starting channels */
17#define SDMA_EVTOVR 0x0010 /* Event override */
18
19/* Set to 0 to prevent channels from ever starting */
20#define SDMA_DSPOVR 0x0014 /* Channel BP override */
21#define SDMA_HOSTOVR 0x0018 /* ARM platform override */
22
23/* Channels which are pending; you can also start channels here;
24 * `done` instruction will clear this.
25 */
26#define SDMA_EVTPEND 0x001C /* Pending events */
27
28/* Bit 0 resets the SDMA; bit 1 forces reschedule as if `done` was done */
29#define SDMA_RESET 0x0024 /* Reset register */
30
31/* New HW request for already pending/running channel; XRUN? */
32#define SDMA_EVTERR 0x0028 /* DMA request error */
33
34#define SDMA_INTRMASK 0x002C /* ARM platform interrupt mask */
35#define SDMA_PSW 0x0030 /* Schedule status */
36
37/* Mirror of SDMA_EVTERR which doesn't clear on reads */
38#define SDMA_EVTERRDBG 0x0034 /* DMA request error register */
39
40#define SDMA_CONFIG 0x0038 /* Configuration register */
41#define SDMA_CONFIG_ACR BIT(4)
42#define SDMA_CONFIG_CSM_MSK MASK(1, 0)
43#define SDMA_CONFIG_CSM_STATIC SET_BITS(1, 0, 0) /* Static ctx switch */
44#define SDMA_CONFIG_CSM_DYN_LP SET_BITS(1, 0, 1) /* Low power; unused */
45#define SDMA_CONFIG_CSM_DYN_NL SET_BITS(1, 0, 2) /* No loops; unused */
46#define SDMA_CONFIG_CSM_DYN SET_BITS(1, 0, 3) /* Fully dynamic ctx switch */
47
48#define SDMA_LOCK 0x003C /* Lock firmware until reset */
49
50/* OnCE debug registers; unsupported, mentioned for completeness */
51/* Should I use these for accessing contexts?! */
52#define SDMA_ONCE_ENB 0x0040 /* Enable OnCE */
53#define SDMA_ONCE_DATA 0x0044 /* OnCE data register */
54#define SDMA_ONCE_INSTR 0x0048 /* OnCE instruction register */
55#define SDMA_ONCE_STAT 0x004C /* OnCE status register */
56#define SDMA_ONCE_CMD 0x0050 /* OnCE command register */
57
58/* Illegal instruction trap handler address */
59#define SDMA_ILLINSTADDR 0x0058
60
61#define SDMA_CHN0ADDR 0x005C /* Channel 0 boot address */
62
63/* Hardware DMA requests 0-47, useful for debug */
64#define SDMA_EVT_MIRROR 0x0060 /* DMA requests */
65#define SDMA_EVT_MIRROR2 0x0064 /* DMA requests */
66
67/* Cross-trigger events configuration registers
68 * Unsupported by this driver
69 */
70#define SDMA_XTRIG_CONF1 0x0070
71#define SDMA_XTRIG_CONF2 0x0074
72
73/* Channel priority registers; priorities 1-7 are useful, 0 means don't
74 * start. Higher number priority is higher priority.
75 */
76#define SDMA_CHNPRI(chan) (0x0100 + 4 * (chan))
77
78#define SDMA_DEFPRI 4
79#define SDMA_MAXPRI 7
80
81#define SDMA_HWEVENTS_COUNT 48
82
83/* Channel to HW mapping; no default values */
84#define SDMA_CHNENBL(hwchan) (0x0200 + 4 * (hwchan))
85
86/* SDMA DONE0/DONE1 configurations; each of 32 events can stop channels
87 * from 0 to 7
88 */
89#define SDMA_DONE0_CONFIG 0x1000
90#define SDMA_DONE1_CONFIG 0x1004
91
92#define SDMA_DONE0_CONFIG_DONE_SEL BIT(7)
93#define SDMA_DONE0_CONFIG_DONE_DIS BIT(6)
94
95#define SDMA_WATERMARK_LEVEL_N_FIFOS(x) SET_BITS(15, 12, x)
96#define SDMA_WATERMARK_LEVEL_OFF_FIFOS(x) SET_BITS(19, 16, x)
97#define SDMA_WATERMARK_LEVEL_WORDS_PER_FIFO(x) SET_BITS(31, 28, x)
98#define SDMA_WATERMARK_LEVEL_SW_DONE BIT(23)
99#define SDMA_WATERMARK_LEVEL_SW_DONE_SEL_OFF 24
100
101/* Buffer descriptor first word */
102/* Count: Data buffer size, in words */
103#define SDMA_BD_COUNT_MASK MASK(15, 0)
104#define SDMA_BD_COUNT(n) SET_BITS(15, 0, n)
105#define SDMA_BD_MAX_COUNT SDMA_BD_COUNT_MASK
106/* Done bit, when 1 SDMA is active */
107#define SDMA_BD_DONE BIT(16)
108/* Wrap bit, last BD in circular buffer */
109#define SDMA_BD_WRAP BIT(17)
110/* Continuous, transfer can bleed into next BD */
111#define SDMA_BD_CONT BIT(18)
112/* Interrupt, should DSP receive interrupt when this BD is complete? */
113#define SDMA_BD_INT BIT(19)
114/* Error (status bit), if some error happened while processing this BD */
115#define SDMA_BD_ERROR BIT(20)
116/* Last: SDMA sets it when there is no more data to transfer */
117#define SDMA_BD_LAST BIT(21)
118/* Bit 22 is reserved */
119/* Bit 23 is undocumented, but used in existing drivers; I think it
120 * means that buf_xaddr contains a valid value
121 */
122#define SDMA_BD_EXTD BIT(23)
123
124/* CMD: Command, differentiator for functionality of scripts; can also
125 * hold error codes returned by SDMA scripts.
126 */
127#define SDMA_BD_CMD_MASK MASK(31, 24)
128#define SDMA_BD_CMD(cmd) SET_BITS(31, 24, cmd)
129
130/* We don't need more than 4 buffer descriptors per channel */
131#define SDMA_MAX_BDS 4
132
133#define SDMA_CMD_C0_SET_PM 0x4
134#define SDMA_CMD_C0_SET_DM 0x1
135#define SDMA_CMD_C0_GET_PM 0x8
136#define SDMA_CMD_C0_GET_DM 0x2
137#define SDMA_CMD_C0_SETCTX(chan) (((chan) << 3) | 7)
138#define SDMA_CMD_C0_GETCTX(chan) (((chan) << 3) | 6)
139/* Used by the actual data transfer scripts, the width of each
140 * elementary transfer; set in command field. Default is 32-bit.
141 */
142#define SDMA_CMD_XFER_SIZE(s) ((s) == 8 ? 1 : (s) == 16 ? 2 : (s) == 24 ? 3 : 0)
143
