208/* Devicelist type
209 * In Reference FW KPB used Bi-dir lists to store modules for FMT. It is possible that lists are
210 * not necessary, but in case it might be wrong, here we use an array
211 * with a list interface to switch it to a list easily.
212 */
213struct device_list {
214 struct comp_dev **devs[DEVICE_LIST_SIZE];
215 /* number of items AND index of next empty box */
216 size_t count;
217};
218
219/*
220 *
221 *
222 * KPB FMT config set steps:
223 * 1. Get dev_ids of module instances from IPC
224 * 2. Alloc this kpb module instance on kpb_list_item, save address of where it was allocated
225 * 3. Push the address on dev_list.device_list
226 * 2. For each dev_id get device handler(module instance)
227 * 3. Alloc device handler in modules_list_item, save address of where it was allocated
228 * 4. Register this address in fmt.device_list
229 *
230 * Pointer structure:
231 *
232 * COMP_DEVS
233 * ^
234 * |
235 * dev_list.modules_list_item(comp_dev* )
236 * ^
237 * |
238 * dev_list.device_list(comp_dev**)
239 * ^
240 * |
241 * fmt.device_list(device_list*)
242 *
243 *
244 */
245
246/* KpbFastModeTaskModulesList Namespace */

249 struct device_list device_list[FAST_MODE_TASK_MAX_LIST_COUNT];
250 /* One for each sinkpin. */
251 struct comp_dev *kpb_list_item[KPB_MAX_SINK_CNT];
252 struct comp_dev *modules_list_item[FAST_MODE_TASK_MAX_MODULES_COUNT];
253 struct comp_dev *kpb_mi_ptr;
254};
