18/* Smart Amplifier component is a two-layer structured design, i.e. generic
19 * layer and inner model layer. The latter can have various implementations
20 * respectively for Amplifier solution suppliers, while the former is the common
21 * part of Smart Amp process adaptable for all solutions.
22 *
23 * In structural aspect, one can regard generic layer as the glue code that
24 * wraps inner model in a SOF component. Ops are defined for interaction between
25 * two layers. Inner model is the solution-specific modular code, which may have
26 * static libraries linked as needed. The structure is figured below:
27 *
28 * SRC(FF) SINK(OUT) +-SRC(FB) bytectl
29 * +- SMART_AMP |^ comp ops | ^ | ^|
30 * | +------------------v|-------------v---------|------v-----------|v---------+
31 * | | Generic Layer (chan remap/fmt conv) || |
32 * | | (memory mgr)--+------> :::::::::BUFFERS::::::::::::: |+> CONFIG |
33 * | +------------------|-|^-----------|---------^------|-----------^|---------+
34 * | | || mod ops | | | ||
35 * | +------------------v-v|-----------v---------|------v-----------|v---------+
36 * | | Inner Model :::::::::::::::::::::BUFFERS::::::::::::::::::::::: |
37 * | | (solution-specific impl/wrapper) |+> MODEL |
38 * | +------------------------------|^------------------------------^----------+
39 * +--- v| lib ops | CALDATA
40 * Static Libs (as needed) ----------+
41 * Note:
42 * - FF(Feed-Forward): un-processed playback frame source
43 * - FB(Feedback): feedback reference frame source (from the capture pipeline)
44 *
45 * As illustrated above, generic layer handles the cross-communication for inner
46 * model and SOF pipeline flow, as well as the smart-amp common tasks including:
47 * 1. Channel remapping for input/output frames
48 * 2. Frame format conversion for input/output frames. It allows inner model to
49 * work with different format from SOF audio stream.
50 * (Now it only allows the bitdepth of inner model format >= SOF stream,
51 * e.g. inner model: S32_LE, SOF stream: S16_LE)
52 * 3. Full-management of runtime memory. That is, dynamic memory buffers either
53 * required by generic layer or inner model will be allocated/owned/released
54 * by generic layer.
55 * More details are stated in the comment of each struct/function later.
56 */

76/* The common base for inner model data structs. Inner model should declare its
77 * model data struct while putting this base as the leading member of struct, e.g.
78 * struct solution_foo_mod_data {
79 * struct smart_amp_mod_data_base base;
80 * uint32_t foo_version;
81 * struct foo_parameter_set;
82 * ...
83 * };
84 *
85 * Then implement mod_data_create() in its own C file, e.g.
86 * struct smart_amp_mod_data_base *mod_data_create(const struct comp_dev *dev)
87 * {
88 * struct solution_foo_mod_data *foo;
89 * foo = rzalloc(SOF_MEM_FLAG_USER, sizeof(*foo));
90 * assert(foo);
91 * foo->base.dev = dev;
92 * foo->base.mod_ops = foo_ops; // declared somewhere as static const
93 * ...
94 * return &foo->base;
95 * }
96 */
97struct smart_amp_mod_data_base {
98 const struct comp_dev *dev; /* for logger tracing use only */
99 const struct inner_model_ops *mod_ops;
100};

108/* For memory allocation, generic layer plays the active role that queries the
109 * required memory size for inner model (then allocate and assign back) in
110 * some specific moments, i.e. once before and after model initialized. Inner
111 * model should consider buffers located and allocated onto 3 memory blocks in
112 * terms of usage:
113 * PRIVATE - allocated before model init - for libraries internal usage
114 * FRAME - allocated after model init - for audio frame buffer usage
115 * PARAM - allocated after model init - for parameter blob usage
116 */
117enum smart_amp_mod_memblk {
118 MOD_MEMBLK_PRIVATE = 0,
119 MOD_MEMBLK_FRAME,
120 MOD_MEMBLK_PARAM,
121 MOD_MEMBLK_MAX
122};
