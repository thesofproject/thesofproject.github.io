58/*
59 * DAI_CONFIG flags. The 4 LSB bits are used for the commands, HW_PARAMS, HW_FREE and PAUSE
60 * representing when the IPC is sent. The 4 MSB bits are used to add quirks along with the above
61 * commands.
62 */
63#define SOF_DAI_CONFIG_FLAGS_CMD_MASK 0xF
64#define SOF_DAI_CONFIG_FLAGS_NONE 0
65#define SOF_DAI_CONFIG_FLAGS_HW_PARAMS BIT(0)
66#define SOF_DAI_CONFIG_FLAGS_HW_FREE BIT(1)
68#define SOF_DAI_CONFIG_FLAGS_PAUSE BIT(2)
69#define SOF_DAI_CONFIG_FLAGS_QUIRK_SHIFT 4
70#define SOF_DAI_CONFIG_FLAGS_QUIRK_MASK (0xF << SOF_DAI_CONFIG_FLAGS_QUIRK_SHIFT)
71/*
72 * This should be used along with the SOF_DAI_CONFIG_FLAGS_HW_PARAMS to indicate that pipeline
73 * stop/pause and DAI DMA stop/pause should happen in two steps. This change is only available
74 * ABI 3.20 onwards.
75 */
76#define SOF_DAI_CONFIG_FLAGS_2_STEP_STOP BIT(0)
77
78#define SOF_DAI_QUIRK_IS_SET(flags, quirk) \
79 (((flags & SOF_DAI_CONFIG_FLAGS_QUIRK_MASK) >> SOF_DAI_CONFIG_FLAGS_QUIRK_SHIFT) & quirk)
