108static inline bool task_is_active(struct task *task)
109{
110 switch (task->state) {
111 case SOF_TASK_STATE_QUEUED:
112 case SOF_TASK_STATE_PENDING:
113 case SOF_TASK_STATE_RUNNING:
114 case SOF_TASK_STATE_PREEMPTED:
115 case SOF_TASK_STATE_RESCHEDULE:
116 return true;
117 default:
118 return false;
119 }
120}
