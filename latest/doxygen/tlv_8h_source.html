85static inline void tlv_value_get(const void *data,
86 uint32_t size,
87 uint32_t type,
88 void **value,
89 uint32_t *length)
90{
91 const struct sof_tlv *tlv = (const struct sof_tlv *)data;
92 const uint32_t end_addr = (uint32_t)data + size;
93
94 while (tlv && (uint32_t)tlv < end_addr) {
95 if (tlv->type == type) {
96 *value = (void *)tlv->value;
97 *length = tlv->length;
98 break;
99 }
100
101 tlv = tlv_next(tlv);
102 }
103}
