63/*
64 * trace_event macro definition
65 *
66 * trace_event() macro is used for logging events that occur at runtime.
67 * It comes in 2 main flavours, atomic and non-atomic. Depending of definitions
68 * above, it might also propagate log messages to mbox if desired.
69 *
70 * First argument is always class of event being logged, as defined in
71 * user/trace.h - TRACE_CLASS_* (deprecated - do not use).
72 * Second argument is string literal in printf format, followed by up to 4
73 * parameters (uint32_t), that are used to expand into string fromat when
74 * parsing log data.
75 *
76 * All compile-time accessible data (verbosity, class, source file name, line
77 * index and string literal) are linked into .static_log_entries section
78 * of binary and then extracted by smex, so they do not contribute to loadable
79 * image size. This way more elaborate log messages are possible and encouraged,
80 * for better debugging experience, without worrying about runtime performance.
81 */

170#define _log_message(ignored_log_func, atomic, level, comp_class, ctx, id_1, id_2, format, ...) \
171do { \
172 (void)ctx; \
173 (void)id_1; \
174 (void)id_2; \
175 struct timeval tv; \
176 char slevel[5][4] = {"dis", "err", "wrn", "inf", "dbg"}; \
177 char *msg = "(%s:%d) " format; \
178 if (level <= host_trace_level) { \
179 gettimeofday(&tv, NULL); \
180 fprintf(stderr, "trace: [%ld.%6.6ld] <%s> ", tv.tv_sec, tv.tv_usec, \
181 slevel[level]); \
182 fprintf(stderr, msg, strrchr(__FILE__, '/') + 1, \
183 __LINE__, ##__VA_ARGS__); \
184 fprintf(stderr, "\n"); \
185 } \
186} while (0)
