78/*
79 * trace_event macro definition
80 *
81 * trace_event() macro is used for logging events that occur at runtime.
82 * It comes in 2 main flavours, atomic and non-atomic. Depending of definitions
83 * above, it might also propagate log messages to mbox if desired.
84 *
85 * First argument is always class of event being logged, as defined in
86 * user/trace.h - TRACE_CLASS_* (deprecated - do not use).
87 * Second argument is string literal in printf format, followed by up to 4
88 * parameters (uint32_t), that are used to expand into string fromat when
89 * parsing log data.
90 *
91 * All compile-time accessible data (verbosity, class, source file name, line
92 * index and string literal) are linked into .static_log_entries section
93 * of binary and then extracted by smex, so they do not contribute to loadable
94 * image size. This way more elaborate log messages are possible and encouraged,
95 * for better debugging experience, without worrying about runtime performance.
96 */

271#define _log_message(ignored_log_func, atomic, level, comp_class, ctx, id_1, id_2, format, ...) \
272do { \
273 (void)ctx; \
274 (void)id_1; \
275 (void)id_2; \
276 struct timeval tv; \
277 char slevel[5][4] = {"dis", "err", "wrn", "inf", "dbg"}; \
278 char *msg = "(%s:%d) " format; \
279 if (level <= host_trace_level) { \
280 gettimeofday(&tv, NULL); \
281 fprintf(stderr, "trace: [%ld.%6.6ld] <%s> ", tv.tv_sec, tv.tv_usec, \
282 slevel[level]); \
283 fprintf(stderr, msg, strrchr(__FILE__, '/') + 1, \
284 __LINE__, ##__VA_ARGS__); \
285 fprintf(stderr, "\n"); \
286 } \
287} while (0)

314do { \
315 _DECLARE_LOG_ENTRY(lvl, format, comp_class, \
316 META_COUNT_VARAGS_BEFORE_COMPILE(__VA_ARGS__)); \
317 STATIC_ASSERT_ARG_SIZE(__VA_ARGS__); \
318 STATIC_ASSERT(_TRACE_EVENT_MAX_ARGUMENT_COUNT >= \
319 META_COUNT_VARAGS_BEFORE_COMPILE(__VA_ARGS__), \
320 BASE_LOG_ASSERT_FAIL_MSG \
321 ); \
322 _log_sofdict(log_func, atomic, &log_entry, ctx, lvl, id_1, id_2, \
323 META_COUNT_VARAGS_BEFORE_COMPILE(__VA_ARGS__), ##__VA_ARGS__); \
324 _log_nodict(atomic, META_COUNT_VARAGS_BEFORE_COMPILE(__VA_ARGS__), \

326} while (0)
