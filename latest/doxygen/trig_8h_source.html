105static inline int32_t sin_fixed_32b(int32_t th_rad_fxp)
106{
107 int32_t th_cdc_fxp;
108 int32_t sign;
109 int32_t b_yn;
110 int32_t xn;
111 cordic_approx(th_rad_fxp, CORDIC_31B_TABLE_SIZE, &sign, &b_yn, &xn, &th_cdc_fxp);
112
113 th_cdc_fxp = sign * b_yn;

115 return sat_int32(Q_SHIFT_LEFT((int64_t)th_cdc_fxp, 30, 31));

301static inline int32_t asin_fixed_32b(int32_t cdc_asin_th)
302{
303 int32_t th_asin_fxp;
304
305 if (cdc_asin_th >= 0)
306 th_asin_fxp = is_scalar_cordic_asin(cdc_asin_th, CORDIC_31B_ITERATIONS);
307 else
308 th_asin_fxp = -is_scalar_cordic_asin(-cdc_asin_th, CORDIC_31B_ITERATIONS);
309
310 return th_asin_fxp; /* Q2.30 */
311}
