99static inline int32_t sin_fixed_32b(int32_t th_rad_fxp)
100{
101 int32_t th_cdc_fxp;

103 int32_t b_yn;
104 int32_t xn;

295static inline int32_t asin_fixed_32b(int32_t cdc_asin_th)
296{
297 int32_t th_asin_fxp;
298
299 if (cdc_asin_th >= 0)
300 th_asin_fxp = is_scalar_cordic_asin(cdc_asin_th, CORDIC_31B_ITERATIONS);
301 else
302 th_asin_fxp = -is_scalar_cordic_asin(-cdc_asin_th, CORDIC_31B_ITERATIONS);
303
304 return th_asin_fxp; /* Q2.30 */
