86 * demands that the symbol name refer to a struct sof_uuid.

92 { .id = initializer, .name = entity_name }; \
93 extern const struct sof_uuid \
94 __attribute__((alias("_" #uuid_name))) uuid_name

97/* XTOS SOF emits two definitions, one into the runtime (which may not

99 * special section via handling in the linker script.

104 __section(".static_uuids") \
