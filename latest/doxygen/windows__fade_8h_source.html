36static inline int32_t volume_pow_175(int32_t val)
37{
38 int32_t y0;
39 int32_t y1;
40 int32_t dx;
41 int32_t c;
42 int32_t y;
43 int i;
44
45 if (val == 0)
46 return 0;
47
48 i = Q_MULTS_32X32((int64_t)val, WINDOWS_FADE_LUT_SIZE - 1, 30, 0, 0);
49 if (i >= WINDOWS_FADE_LUT_SIZE - 1)
50 return Q_CONVERT_FLOAT(1, 30);
51
52 y0 = windows_fade_lookup[i];
53 y1 = windows_fade_lookup[i + 1];
54 c = (y1 - y0) * (WINDOWS_FADE_LUT_SIZE - 1); /* Q1.15 */
55 dx = val - i * WINDOWS_FADE_LUT_STEP; /* Q2.30 */
56 y = Q_SHIFT_LEFT(y0, 15, 30) + Q_MULTSR_32X32((int64_t)dx, c, 30, 15, 30);
57 return y;
58}
