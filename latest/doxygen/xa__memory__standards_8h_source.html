38/* standard memory types */
39/* to be used inter frames */
40#define XA_MEMTYPE_PERSIST 0x00
41/* read write, to be used intra frames */
42#define XA_MEMTYPE_SCRATCH 0x01
43/* read only memory, intra frame */
44#define XA_MEMTYPE_INPUT 0x02
45/* read-write memory, for usable output, intra frame */
46#define XA_MEMTYPE_OUTPUT 0x03

48#define XA_MEMTYPE_TABLE 0x04
49/* input buffer before mem tabs allocation */
50#define XA_MEMTYPE_PRE_FRAME_INPUT 0x05

52#define XA_MEMTYPE_PRE_FRAME_SCRATCH 0x06
53/* for local variables */
