34/* The fallthrough attribute is supported since GCC 7.0
35 * and Clang 10.0.0.
36 *
37 * Note that Clang sets __GNUC__ == 4 so the GCC version
38 * test will not be true here, and must go through
39 * the Clang version test.
40 */
41#if ((defined(__GNUC__) && (__GNUC__ >= 7)) || \
42 (defined(__clang__) && (__clang_major__ >= 10))) && !defined(__CHECKER__)

45#define COMPILER_FALLTHROUGH /* fallthrough */
