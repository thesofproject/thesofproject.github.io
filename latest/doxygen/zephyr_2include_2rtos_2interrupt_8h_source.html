27static inline int interrupt_register(uint32_t irq, void(*handler)(void *arg), void *arg)
28{
29#ifdef CONFIG_DYNAMIC_INTERRUPTS
30 return arch_irq_connect_dynamic(irq, 0, (void (*)(const void *))handler,
31 arg, 0);
32#else
33 LOG_MODULE_DECLARE(zephyr, CONFIG_SOF_LOG_LEVEL);
34 tr_err(&zephyr_tr, "Cannot register handler for IRQ %u: dynamic IRQs are disabled",
35 irq);
36 return -EOPNOTSUPP;
37#endif
38}
